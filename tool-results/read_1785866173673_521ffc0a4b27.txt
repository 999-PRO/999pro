    84→    84→    84→ * CORS origin checker.
    85→    85→    85→ *
    86→    86→    86→ * v25.2-CORS-FIX: When origin is NOT allowed, we now pass an Error to the
    87→    87→    87→ * callback (callback(err)) instead of callback(null, false). This makes
    88→    88→    88→ * cors() respond with 500 + error message instead of passing the request
    89→    89→    89→ * through to Express (which would respond 200 + Allow header without any
    90→    90→    90→ * Access-Control-Allow-Origin, confusing the browser).
    91→    91→    91→ *
    92→    92→    92→ * In practice this shouldn't matter anymore because the frontend now uses
    93→    93→    93→ * relative /api/* paths (no cross-origin), but this is defense-in-depth:
    94→    94→    94→ * if someone manually calls the backend from a different origin, they get
    95→    95→    95→ * a clear error instead of a silent 200-without-CORS-header.
    96→    96→    96→ *
    97→    97→    97→ * In development: allow any origin on localhost or LAN IPs (so the app works
    98→    98→    98→ * when opened from a phone at http://192.168.x.x:3000), plus any *.space-z.ai
    99→    99→    99→ * subdomain (the sandbox/preview gateway) and the wildcard `*` if configured.
   100→   100→   100→ *
   101→   101→   101→ * In production: only allow the configured CLIENT_ORIGIN allowlist
   102→   102→   102→ * (or everything if CLIENT_ORIGIN contains `*`).
   103→   103→   103→ */
   104→   104→   104→function corsOrigin(origin: string | undefined, callback: (err: Error | null, ok?: boolean) => void) {
   105→   105→   105→  // Allow same-origin requests (no Origin header, e.g. curl, server-to-server)
   106→   106→   106→  if (!origin) return callback(null, true)
   107→   107→   107→
   108→   108→   108→  // Wildcard shortcut: if CLIENT_ORIGIN contains `*`, allow every origin.
   109→   109→   109→  // In production this is REJECTED — wildcards are dev-only.
   110→   110→   110→  if (CLIENT_ORIGIN.includes('*')) {
   111→   111→   111→    if (isProd) {
   112→   112→   112→      logger.warn('CORS CLIENT_ORIGIN=* forbidden in production', { module: 'cors', origin })
   113→   113→   113→      return callback(new Error(`CORS: wildcard origin forbidden in production (got ${origin})`))
   114→   114→   114→    }
   115→   115→   115→    return callback(null, true)
   116→   116→   116→  }
   117→   117→   117→
   118→   118→   118→  // Production: strict allowlist
   119→   119→   119→  if (isProd) {
   120→   120→   120→    if (CLIENT_ORIGIN.includes(origin)) return callback(null, true)
   121→   121→   121→    // Also allow any *.space-z.ai preview subdomain in prod (sandbox gateway)
   122→   122→   122→    try {
   123→   123→   123→      const url = new URL(origin)
   124→   124→   124→      if (url.hostname === 'space-z.ai' || url.hostname.endsWith('.space-z.ai')) {
   125→   125→   125→        return callback(null, true)
   126→   126→   126→      }
   127→   127→   127→    } catch {}
   128→   128→   128→    logger.warn('CORS origin rejected (production allowlist)', {
   129→   129→   129→      module: 'cors',
   130→   130→   130→      origin,
   131→   131→   131→      allowlist: CLIENT_ORIGIN,
   132→   132→   132→    })
   133→   133→   133→    return callback(new Error(`CORS: origin ${origin} not in CLIENT_ORIGIN allowlist`))
   134→   134→   134→  }
   135→   135→   135→
   136→   136→   136→  // Development: allow localhost + LAN IPs on any port
   137→   137→   137→  try {
   138→   138→   138→    const url = new URL(origin)
   139→   139→   139→    const host = url.hostname
   140→   140→   140→    if (
   141→   141→   141→      host === 'localhost' ||
   142→   142→   142→      host === '127.0.0.1' ||
   143→   143→   143→      /^(192\.168\.|10\.|172\.(1[6-9]|2[0-9]|3[01])\.|169\.254\.)/.test(host) ||
   144→   144→   144→      host.endsWith('.local')
   145→   145→   145→    ) {
   146→   146→   146→      return callback(null, true)
   147→   147→   147→    }
   148→   148→   148→    // Sandbox preview gateway: any *.space-z.ai subdomain (e.g. preview-<bot-id>.space-z.ai)
   149→   149→   149→    if (host === 'space-z.ai' || host.endsWith('.space-z.ai')) {
   150→   150→   150→      return callback(null, true)
   151→   151→   151→    }
   152→   152→   152→  } catch {}
   153→   153→   153→  // Fallback: allow configured origins
   154→   154→   154→  if (CLIENT_ORIGIN.includes(origin)) return callback(null, true)
   155→   155→   155→  logger.warn('CORS origin rejected (development)', {
   156→   156→   156→    module: 'cors',
   157→   157→   157→    origin,
   158→   158→   158→    allowlist: CLIENT_ORIGIN,
   159→   159→   159→  })
   160→   160→   160→  return callback(new Error(`CORS: origin ${origin} not allowed`))
   161→   161→   161→}
