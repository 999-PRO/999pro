   350→        // GET /api/auth/verify-email and shows a success screen. The old
   351→        // implementation called the API endpoint directly — which returned
   352→        // raw JSON and was a poor UX. The new frontend page wraps the same

   354→        const verifyUrl = `${baseUrl}/verify-email?token=${verifyToken}`
   355→        const { sendVerificationEmail } = await import('../lib/email.js')
   356→        // Fire-and-forget — email failures must not break registration.

   358→          logger.error('[register] Auto-send verification email failed:', {
   359→            module: 'auth',
   360→            error: e,
   361→            userId: user.id,
   362→          })
   363→        })
   364→      } catch (e) {
   365→        // Non-fatal: the user can still request a new verification email.
   366→        logger.error('[register] Failed to queue verification email:', {
   367→          module: 'auth',
   368→          error: e,
   369→          userId: user.id,
   370→        })
   371→      }
   372→    }
   373→
   374→    res.status(201).json({ token, user: publicUser(user, { includeContact: true }) })
   375→  }),

   378→// POST /api/auth/login — login field accepts email | username | phone
   379→router.post(
   380→  '/login',
   381→  asyncHandler(async (req, res) => {
   382→    const { login: rawLogin, password } = loginSchema.parse(req.body)
   383→    const login = normaliseLogin(rawLogin)
   384→
   385→    // Try by email or username (both lowercased), then phone
   386→    // Phase 8.1: filter out soft-deleted users — they cannot log in.
   387→    const user = await prisma.user.findFirst({
   388→      where: {
   389→        deletedAt: null,
   390→        OR: [
   391→          { email: login },
   392→          { username: login },
   393→          ...(login.startsWith('+') || /^\d/.test(login) ? [{ phone: rawLogin.trim() }] : []),
   394→        ],
   395→      },
   396→    })
   397→
   398→    // S-CRIT-009 fix: account lockout after 5 failed attempts (15 min).
   399→    if (user?.lockedUntil && new Date(user.lockedUntil) > new Date()) {
   400→      const remainingMs = new Date(user.lockedUntil).getTime() - Date.now()
   401→      const remainingMin = Math.ceil(remainingMs / 60000)
   402→      await auditLogRaw(user.id, req, 'auth', user.id, 'login_failed', {
   403→        after: { login, reason: 'account_locked', remainingMin },
   404→      })
   405→      return res.status(429).json({
   406→        error: `Аккаунт заблокирован. Попробуйте через ${remainingMin} мин.`,
   407→        lockedUntil: user.lockedUntil,
   408→      })
   409→    }
   410→
   411→    // Use safe compare to mitigate timing oracle for user enumeration
   412→    const ok = await safeComparePassword(password, user)
   413→
   414→    if (!user || !ok) {
   415→      // S-CRIT-009: increment failedLoginCount. Lock for 15 min on 5th failure.
   416→      if (user) {
   417→        const newCount = (user.failedLoginCount || 0) + 1
   418→        const shouldLock = newCount >= 5
   419→        await prisma.user.update({
   420→          where: { id: user.id },
   421→          data: {
   422→            failedLoginCount: newCount,
   423→            ...(shouldLock ? { lockedUntil: new Date(Date.now() + 15 * 60 * 1000) } : {}),
   424→          },
   425→        })
   426→      }
   427→      // Phase 32: audit log failed login attempt (for brute-force detection)
   428→      await auditLogRaw(user?.id || null, req, 'auth', user?.id || null, 'login_failed', {
   429→        after: { login: login, reason: !user ? 'user_not_found' : 'wrong_password', attempt: (user?.failedLoginCount || 0) + 1 },
   430→      })
   431→      const msg = user && (user.failedLoginCount || 0) + 1 >= 5
   432→        ? 'Слишком много неудачных попыток. Аккаунт заблокирован на 15 минут.'
   433→        : 'Invalid credentials'
   434→      return res.status(401).json({ error: msg })
   435→    }
   436→
   437→    // S-CRIT-009: reset failedLoginCount on successful login
   438→    if (user.failedLoginCount > 0 || user.lockedUntil) {
   439→      await prisma.user.update({
   440→        where: { id: user.id },
   441→        data: { failedLoginCount: 0, lockedUntil: null },
   442→      })
   443→    }
   444→
   445→    // v9-audit-fix: S-CRIT-008 — 2FA/TOTP verification for enabled accounts.
   446→    // If the user has TOTP enabled, require a valid 6-digit code before
   447→    // issuing a JWT. If no code is provided, return a "totpRequired" flag
   448→    // so the frontend can show the TOTP input form.
   449→    // v19.0: Also supports:
   450→    //   - backup codes (8-char alphanumeric) — consumed on use
   451→    //   - email 2FA code (6-digit) — if user.emailTwoFactorCode is set
   452→    if (user.totpEnabled && user.totpSecret) {
   453→      const totpCode = (req.body as { totpCode?: string })?.totpCode
   454→      if (!totpCode) {
   455→        // Don't reveal whether the password was correct — but we already
   456→        // reset the failedLoginCount above. That's OK because the password
   457→        // WAS correct; we're just asking for the second factor.
   458→        return res.json({ totpRequired: true, user: publicUser(user, { includeContact: true }) })
   459→      }
   460→      const { verifyTotp } = await import('../lib/totp.js')
   461→      const totpOk = verifyTotp(totpCode, user.totpSecret)
   462→      // v19.0: try backup code if TOTP didn't match
   463→      const backupOk = totpOk ? false : await tryBackupCode(user.id, totpCode.toUpperCase())
   464→      if (!totpOk && !backupOk) {
   465→        await auditLogRaw(user.id, req, 'auth', user.id, 'login_failed', {
   466→          after: { reason: 'invalid_totp' },
   467→        })
   468→        return res.status(401).json({ error: 'Неверный код подтверждения.' })
   469→      }
   470→      await auditLogRaw(user.id, req, 'auth', user.id, 'totp_used', {
   471→        after: { method: backupOk ? 'backup_code' : 'totp' },
   472→      })
   473→    }
   474→
   475→    // Wave 3 (S-PROD-001): mandatory TOTP setup for admin role.
   476→    // Admins who haven't enabled TOTP yet are forced to set it up before
   477→    // they can access the admin panel. Returns `totpSetupRequired: true`
   478→    // so the frontend (Studio) shows the setup wizard. The password is
   479→    // correct (verified above) — we just gate the regular JWT behind 2FA
   480→    // enrollment.
   481→    //
   482→    // ARCHITECTURAL FIX (was: "Authorization header missing" bug):
   483→    // Previously this branch returned NO token, but `/totp/setup` and
   484→    // `/totp/verify` use `requireAuth` — so the client had no way to
   485→    // authenticate the setup calls. Now we issue a SHORT-LIVED
   486→    // `totpPending: true` setup token (15 min) that:
   487→    //   • is accepted by `requireAuth` (so /totp/setup, /totp/verify work)
   488→    //   • is REJECTED by `requireAdmin` (so no admin endpoint can be
   489→    //     accessed until TOTP enrollment completes)
   490→    // After `/totp/verify` succeeds, it issues a fresh regular JWT that
   491→    // the client swaps in for the setup token — full access restored.
   492→    //
   493→    // S-HIGH-001 fix: TOTP enforcement for admins is now UNCONDITIONAL.
   494→    // Previously gated by `process.env.NODE_ENV === 'production'`, which let
   495→    // staging/preview/sandbox admins skip 2FA entirely — risky if a staging
   496→    // DB is ever promoted to prod, or if prod briefly runs with NODE_ENV
   497→    // unset during debugging. Now admins must enroll TOTP in every env.
   498→    // For local dev where 2FA is inconvenient, set BYPASS_ADMIN_TOTP=true

   543→// ============================================================================
   544→// H2 fix: POST /api/auth/refresh — exchange a refresh token for a new access
   545→// token. The refresh token is sent in the request body (or as a Bearer token
   546→// in the Authorization header — clients that already use that header for
   547→// access tokens can re-use the same plumbing).
   548→//
   549→// Flow:
   550→//   1. Client's access token expires (15min in prod, 7d in dev).
   551→//   2. Client calls POST /api/auth/refresh with { refreshToken: "..." }.
   552→//   3. Server verifies the refresh token signature + checks AppSetting for
   553→//      revocation. If valid, issues a fresh access token.
   554→//   4. Client swaps the old access token for the new one and retries the
   555→//      original request.
   556→//
   557→// The refresh token itself is NOT rotated on each refresh (simpler UX).
   558→// It's revoked only on explicit logout or password change.
   559→// ============================================================================
   560→router.post(
   561→  '/refresh',
   562→  asyncHandler(async (req, res) => {
   563→    const refreshToken =
   564→      (req.body as { refreshToken?: string })?.refreshToken ||
   565→      (req.headers.authorization?.startsWith('Bearer ')
   566→        ? req.headers.authorization.slice(7)
   567→        : undefined)
   568→
   569→    if (!refreshToken) {
   570→      return res.status(401).json({ error: 'Refresh token required.' })
   571→    }
   572→
   573→    try {
   574→      const {
   575→        signRefreshToken,
   576→        verifyRefreshToken,
   577→      } = await import('../lib/auth.js')
   578→      const payload = await verifyRefreshToken(refreshToken)
   579→
   580→      // Look up the user — they may have been deleted or had their
   581→      // tokenVersion bumped since the refresh token was issued.
   582→      const user = await prisma.user.findUnique({
   583→        where: { id: payload.sub, deletedAt: null },
   584→        select: { id: true, username: true, role: true, tokenVersion: true },
   585→      })
   586→      if (!user) {
   587→        return res.status(401).json({ error: 'User not found.' })
   588→      }
   589→      if (user.tokenVersion !== payload.v) {
   590→        return res.status(401).json({ error: 'Token version mismatch — please log in again.' })
   591→      }
   592→
   593→      // Issue a fresh access token.
   594→      const access = signToken({
   595→        sub: user.id,
   596→        username: user.username,
   597→        role: user.role as 'user' | 'admin',
   598→        v: user.tokenVersion,
   599→      })
   600→
   601→      // Also issue a fresh refresh token (rotate the jti so the old one
   602→      // becomes invalid — defense against replay attacks). Revoke the old
   603→      // jti and persist the new one.
   604→      const { token: newRefresh, jti: newJti } = signRefreshToken({
   605→        sub: user.id,
   606→        username: user.username,
   607→        role: user.role as 'user' | 'admin',
   608→        v: user.tokenVersion,
   609→      })
   610→      const { persistRefreshToken, revokeRefreshToken } = await import('../lib/auth.js')
   611→      await revokeRefreshToken(payload.jti)
   612→      await persistRefreshToken(newJti, user.id)
   613→
   614→      res.json({ token: access, refreshToken: newRefresh })
   615→    } catch (e) {
   616→      return res.status(401).json({ error: 'Invalid or expired refresh token.' })
   617→    }
   618→  }),

   772→// POST /api/auth/setup-admin  — public, but ONLY works if NO admin exists yet.
   773→// Creates the first admin account and returns a JWT (auto-login).
   774→//
   775→// SECURITY MODEL (v25 — web-first setup):
   776→//   The previous version required a one-time `FIRST_RUN_TOKEN` env var and
   777→//   an `X-Setup-Admin-Token` HTTP header. This forced the operator to run
   778→//   `curl` from the server shell to create the first admin — exactly the
   779→//   kind of CLI step the new "web-only setup" UX is designed to eliminate.
   780→//
   781→//   The endpoint is now token-less. Protection against abuse comes from:
   782→//     1. The `adminCount === 0` precondition — once any admin exists, the
   783→//        endpoint hard-returns 403 forever. An attacker who reaches it
   784→//        after setup completes gets nothing.
   785→//     2. The authLimiter (20 / 15 min / IP) mounted on /api/auth/*.
   786→//     3. A transactional count + create so two concurrent first-time
   787→//        requests cannot both create admins (B-HIGH-009 race fix).
   788→//     4. Username + email uniqueness checks (handled by Prisma P2002 too).
   789→//
   790→//   The remaining risk window is: between server boot and the operator
   791→//   completing the wizard, anyone who can reach `/api/auth/setup-admin`
   792→//   can become the first admin. This is acceptable for the same reason
   793→//   it's acceptable on every CMS that ships a /install endpoint (WordPress,
   794→//   Drupal, Nextcloud, …): the operator is expected to bring the server up
   795→//   behind a private network or reverse proxy, complete setup, and only
   796→//   then expose it publicly. The README documents this explicitly.
   797→router.post(
   798→  '/setup-admin',
   799→  asyncHandler(async (req, res) => {
   800→    // Block if an admin already exists — prevents privilege escalation.
   801→    // Phase 8.1: only count active (non-deleted) admins.
   802→    // The count + create run inside a single $transaction so two concurrent
   803→    // first-time requests cannot both observe adminCount===0 (B-HIGH-009).
   804→    const existingCount = await prisma.user.count({ where: { role: 'admin', deletedAt: null } })
   805→    if (existingCount > 0) {
   806→      return res
   807→        .status(403)
   808→        .json({ error: 'Setup already completed. Use the regular login.', code: 'setup_completed' })
   809→    }
   810→
   811→    const { createAdminSchema } = await import('../lib/schemas.js')
   812→    const data = createAdminSchema.parse(req.body)
   813→
   814→    if (data.password !== data.confirmPassword) {
   815→      return res.status(400).json({ error: 'Passwords do not match' })
   816→    }
   817→
   818→    // v19.0: validate password against DB-configured SecuritySettings.
   819→    // (On a fresh install SecuritySettings doesn't exist yet → defaults apply:
   820→    // min 8 chars, no complexity requirements. The wizard also enforces
   821→    // client-side validation, but server-side is authoritative.)
   822→    const pwdCheck = await validatePasswordAgainstSettings(data.password)
   823→    if (!pwdCheck.ok) {
   824→      return res.status(400).json({ error: pwdCheck.errors.join('. '), errors: pwdCheck.errors })
   825→    }
   826→
   827→    const email = data.email.trim().toLowerCase()
   828→    const username = data.username.trim().toLowerCase()
   829→
   830→    // Uniqueness pre-check — gives a clearer error than the generic
   831→    // register endpoint. If an existing *non-admin* user already holds the
   832→    // requested email/username, the operator must pick different
   833→    // credentials (or use the reset-admin flow).
   834→    const clash = await prisma.user.findFirst({
   835→      where: { OR: [{ email }, { username }] },
   836→      select: { id: true, email: true, username: true, role: true },
   837→    })
   838→    if (clash) {
   839→      if (clash.role === 'admin') {
   840→        // Should never reach here — we already blocked above when
   841→        // existingCount > 0. Keep the defensive message just in case.
   842→        return res
   843→          .status(409)
   844→          .json({ error: 'Администратор с таким email или логином уже существует. Используйте форму входа.' })
   845→      }
   846→      return res.status(409).json({
   847→        error:
   848→          'Этот email или логин уже занят обычным пользователем. ' +
   849→          'Используйте другой email/логин, либо войдите под этим аккаунтом и затем сбросьте админа ' +
   850→          '(кнопка «Забыли пароль? Сбросить админа» в Studio) с использованием reset-токена.',
   851→      })
   852→    }
   853→
   854→    const password = await hashPassword(data.password)
   855→
   856→    // Transactional create — re-checks adminCount inside the transaction
   857→    // so a concurrent setup-admin request that committed between our
   858→    // outer count and our create cannot also create an admin.
   859→    const user = await prisma.$transaction(async (tx) => {
   860→      const adminCount = await tx.user.count({ where: { role: 'admin', deletedAt: null } })
   861→      if (adminCount > 0) {
   862→        // Defensive: a concurrent request beat us to it. Surface a 409
   863→        // without leaking details.
   864→        throw new SetupAlreadyCompletedError()
   865→      }
   866→      return tx.user.create({
   867→        data: {
   868→          email,
   869→          username,
   870→          password,
   871→          displayName: data.displayName,
   872→          role: 'admin',
   873→        },
   874→      })
   875→    }).catch((err) => {
   876→      if (err instanceof SetupAlreadyCompletedError) {
   877→        return null
   878→      }
   879→      throw err
   880→    })
   881→
   882→    if (!user) {
   883→      // Concurrent request won the race.
   884→      return res
   885→        .status(409)
   886→        .json({ error: 'Setup already completed.', code: 'setup_completed' })
   887→    }
   888→
   889→    const token = signToken({
   890→      sub: user.id,
   891→      username: user.username,
   892→      role: 'admin',
   893→      v: user.tokenVersion,
   894→    })

   897→      after: { username: user.username, email: user.email, note: 'First admin created via setup-admin (web wizard)' },
   898→    })

   900→  }),

  1295→/**
  1296→ * Try to authenticate using a TOTP backup code.
  1297→ * Backup codes are stored as bcrypt hashes. On successful match, the
  1298→ * code is removed from the user's list (one-time use).
  1299→ * Returns true if a backup code matched.
  1300→ */
  1301→async function tryBackupCode(userId: string, code: string): Promise<boolean> {
  1302→  const user = await prisma.user.findUnique({
  1303→    where: { id: userId },
  1304→    select: { totpBackupCodes: true },
  1305→  })
  1306→  if (!user?.totpBackupCodes) return false
  1307→  let arr: string[] = []
  1308→  try {
  1309→    arr = JSON.parse(user.totpBackupCodes)
  1310→    if (!Array.isArray(arr)) return false
  1311→  } catch {
  1312→    return false
  1313→  }
  1314→  for (let i = 0; i < arr.length; i++) {
  1315→    const hash = arr[i]
  1316→    try {
  1317→      const match = await bcrypt.compare(code, hash)
  1318→      if (match) {
  1319→        // Remove this code (one-time use)
  1320→        arr.splice(i, 1)
  1321→        await prisma.user.update({
  1322→          where: { id: userId },
  1323→          data: { totpBackupCodes: JSON.stringify(arr) },
  1324→        })
  1325→        return true
  1326→      }
  1327→    } catch {
  1328→      // skip malformed hashes
  1329→    }
  1330→  }
  1331→  return false
  1332→}

  1475→      const baseUrl = process.env.APP_PUBLIC_URL || 'http://localhost:3000'

  1491→// POST /api/auth/reset-password — set a new password using a reset token.
  1492→// (Rate-limited globally by app.use('/api/auth', authLimiter) in index.ts.)
  1493→router.post(
  1494→  '/reset-password',
  1495→  asyncHandler(async (req, res) => {
  1496→    const schema = z.object({
  1497→      token: z.string().min(64).max(64),
  1498→      newPassword: z.string().min(8).max(200),
  1499→    })
  1500→    const parsed = schema.safeParse(req.body)
  1501→    if (!parsed.success) {
  1502→      return res.status(400).json({ error: 'Неверный запрос', issues: parsed.error.issues })
  1503→    }
  1504→    const { token, newPassword } = parsed.data
  1505→
  1506→    // Look up the token in AppSetting.
  1507→    const setting = await prisma.appSetting.findUnique({ where: { id: `pwd:reset:${token}` } })
  1508→    if (!setting) {
  1509→      return res.status(400).json({ error: 'Недействительный или истёкший токен' })
  1510→    }
  1511→
  1512→    let tokenData: { userId: string; expires: string }
  1513→    try {
  1514→      tokenData = JSON.parse(setting.value)
  1515→    } catch {
  1516→      return res.status(400).json({ error: 'Недействительный токен' })
  1517→    }
  1518→
  1519→    // Check expiry.
  1520→    if (new Date(tokenData.expires) < new Date()) {
  1521→      // Delete the expired token so it can't be reused.
  1522→      await prisma.appSetting.delete({ where: { id: `pwd:reset:${token}` } }).catch(() => {})
  1523→      return res.status(400).json({ error: 'Срок действия токена истёк. Запросите новый.' })
  1524→    }
  1525→
  1526→    // Fetch the user.
  1527→    const user = await prisma.user.findUnique({
  1528→      where: { id: tokenData.userId, deletedAt: null },
  1529→      select: { id: true, username: true, tokenVersion: true },
  1530→    })
  1531→    if (!user) {
  1532→      return res.status(400).json({ error: 'Аккаунт не найден' })
  1533→    }
  1534→
  1535→    // Hash the new password + bump tokenVersion to invalidate ALL existing
  1536→    // sessions (including the attacker's, if the password was leaked).
  1537→    const newHash = await hashPassword(newPassword)
  1538→    await prisma.$transaction([
  1539→      prisma.user.update({
  1540→        where: { id: user.id },
  1541→        data: {
  1542→          password: newHash,
  1543→          tokenVersion: { increment: 1 }, // invalidate all existing JWTs
  1544→          failedLoginCount: 0,
  1545→          lockedUntil: null, // clear any active lockout
  1546→        },
  1547→      }),
  1548→      // Delete the reset token so it can't be reused.
  1549→      prisma.appSetting.delete({ where: { id: `pwd:reset:${token}` } }),
  1550→    ])
  1551→
  1552→    // Audit the successful reset.
  1553→    await auditLogRaw(user.id, req, 'auth', user.id, 'password_reset_completed', {
  1554→      after: { username: user.username },
  1555→    })
  1556→
  1557→    // Kick any active socket sessions so the user has to re-login everywhere.
  1558→    try {
  1559→      const { kickUserSockets } = await import('../socket/handlers.js')
  1560→      kickUserSockets(user.id)
  1561→    } catch { /* non-critical */ }
  1562→
  1563→    res.json({ ok: true, message: 'Пароль изменён. Войдите с новым паролем.' })
  1564→  }),
  1565→)
