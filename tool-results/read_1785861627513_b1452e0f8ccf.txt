    36→    36→ * Send a Web Push notification to ALL participants of a conversation
    37→    37→ * (except the sender).
    38→    38→ *
    39→    39→ * v24.5 BUGFIX: Previously only sent push to OFFLINE participants (those
    40→    40→ * with no active socket connections). But "online" (socket connected) does
    41→    41→ * NOT mean "looking at the app" — the tab could be in the background. In
    42→    42→ * that case the user got the real-time socket message but NO system
    43→    43→ * notification, so they missed messages when the tab was minimized.
    44→    44→ *
    45→    45→ * Now: send push to ALL participants. The Service Worker decides:
    46→    46→ *   • If the app tab is VISIBLE → suppress the system notification (forward
    47→    47→ *     to the page for in-app toast)
    48→    48→ *   • If the app tab is HIDDEN or CLOSED → show a system notification
    49→    49→ *
    50→    50→ * This ensures the user ALWAYS gets notified, whether the app is open,
    51→    51→ * minimized, or fully closed.
    52→    52→ *
    53→    53→ * Called from BOTH the socket message:send handler AND the REST POST
    54→    54→ * /api/chat/conversations/:id/messages endpoint so the notification policy
    55→    55→ * is consistent regardless of how the message was sent.
    56→    56→ */
    57→    57→export async function notifyOfflineParticipants(
    58→    58→  conversationId: string,
    59→    59→  senderId: string,
    60→    60→  message: { id?: string; content?: string | null; mediaType?: string | null; attachments?: any[] | null; sender?: { displayName?: string | null; username?: string } | null },
    61→    61→): Promise<void> {
    62→    62→  try {
    63→    63→    const participants = await prisma.conversationParticipant.findMany({
    64→    64→      where: { conversationId, userId: { not: senderId } },
    65→    65→      select: { userId: true },
    66→    66→    })
    67→    67→
    68→    68→    // v24.5: send push to ALL participants (not just offline).
    69→    69→    // The SW handles suppression when the app is visible.
    70→    70→    const recipientIds = participants.map((p) => p.userId)
    71→    71→
    72→    72→    if (recipientIds.length === 0) return
    73→    73→
    74→    74→    const senderName = message.sender?.displayName || message.sender?.username || 'Пользователь'
    75→    75→    // v12: use shared buildMessagePreview which handles attachments
    76→    76→    const body = buildMessagePreview({
    77→    77→      content: message.content,
    78→    78→      mediaType: message.mediaType,
    79→    79→      attachments: message.attachments,
    80→    80→    })
    81→    81→
    82→    82→    // UNIQUE TAG PER MESSAGE — previously used `msg-${conversationId}` which
    83→    83→    // caused the OS to replace previous notifications from the same
    84→    84→    // conversation. A user receiving 5 rapid messages only saw the last
    85→    85→    // one. Now each message has its own tag (`msg-${conversationId}-${messageId}`)
    86→    86→    // and renotify:true so each one alerts the user. The OS shade groups
    87→    87→    // them by conversation automatically.
    88→    88→    const messageId = message.id || `${Date.now()}`
    89→    89→    const tag = `msg-${conversationId}-${messageId}`
    90→    90→
    91→    91→    for (const recipientId of recipientIds) {
    92→    92→      void sendPushToUser(recipientId, {
    93→    93→        title: senderName,
    94→    94→        body,
    95→    95→        tag,
    96→    96→        url: '/?view=chat',
    97→    97→        data: { conversationId, messageId },
    98→    98→        renotify: true,
    99→    99→      }).catch((e) => {
   100→   100→        // Log push failures so we can diagnose "user didn't get notified" issues.
   101→   101→        // Previously this was silently swallowed — made debugging impossible.
   102→   102→        console.error('[PUSH] notifyOfflineParticipants: sendPushToUser failed for', recipientId, e instanceof Error ? e.message : e)
   103→   103→      })
   104→   104→    }
   105→   105→  } catch (e) {
   106→   106→    // Push is best-effort — never fail the message send because of a push error.
   107→   107→    // But log it so we can diagnose "user didn't get notified" issues.
   108→   108→    console.error('[PUSH] notifyOfflineParticipants error:', e instanceof Error ? e.message : e)
   109→   109→  }
   110→   110→}

   147→   147→async function tryAuth(socket: Socket): Promise<{ userId: string; username: string } | null> {
   148→   148→  const token =
   149→   149→    (socket.handshake.auth?.token as string | undefined) ||
   150→   150→    (socket.handshake.headers.authorization?.replace(/^Bearer\s+/i, '') as string | undefined)
   151→   151→  if (!token) return null
   152→   152→  try {
   153→   153→    const decoded: JwtPayload = verifyToken(token)
   154→   154→    // B-HIGH-008: short-circuit the DB lookup when the user's auth entry is
   155→   155→    // already cached by a recent REST request. Without this, every socket
   156→   156→    // connect (initial, reconnect, multiple tabs, mobile background→foreground
   157→   157→    // transitions) triggered a `prisma.user.findUnique` — under load this
   158→   158→    // produced thousands of identical queries per second on the same user ids
   159→   159→    // that had JUST been validated by requireAuth on the bootstrap HTTP
   160→   160→    // request. The cache invalidates automatically on token-version bump
   161→   161→    // (getCachedAuth compares v and deletes on mismatch).
   162→   162→    const cached = getCachedAuth(decoded.sub, decoded.v)
   163→   163→    if (cached) {
   164→   164→      return { userId: decoded.sub, username: decoded.username }
   165→   165→    }
   166→   166→    // Cache miss — fall through to DB lookup, then populate the cache so the
   167→   167→    // NEXT socket connect (and any REST request) hits the cache.
   168→   168→    const user = await prisma.user.findUnique({
   169→   169→      where: { id: decoded.sub },
   170→   170→      select: { tokenVersion: true, deletedAt: true, role: true },
   171→   171→    })
   172→   172→    if (!user || user.deletedAt) return null
   173→   173→    if (user.tokenVersion !== decoded.v) return null
   174→   174→    setCachedAuth(decoded.sub, {
   175→   175→      v: user.tokenVersion,
   176→   176→      role: user.role as 'user' | 'admin',
   177→   177→      expires: Date.now() + 60_000,
   178→   178→    })
   179→   179→    return { userId: decoded.sub, username: decoded.username }
   180→   180→  } catch {
   181→   181→    return null
   182→   182→  }
   183→   183→}
   184→   184→
   185→   185→/**
   186→   186→ * Per-socket rate limiter — token bucket per event name.
   187→   187→ * Limits how fast a single client can emit any given event.
   188→   188→ *
   189→   189→ * Without this, a single malicious user can flood message:send or
   190→   190→ * typing:start at thousands of events per second, each triggering a DB
   191→   191→ * write + push dispatch + room broadcast.
   192→   192→ */
   193→   193→interface RateBucket { tokens: number; lastRefill: number }
   194→   194→const RATE_LIMITS: Record<string, { capacity: number; refillPerSec: number }> = {
   195→   195→  'message:send': { capacity: 8, refillPerSec: 2 }, // 8 burst, then 2/sec
   196→   196→  'message:forward': { capacity: 5, refillPerSec: 1 },
   197→   197→  'typing:start': { capacity: 5, refillPerSec: 2 },
   198→   198→  'message:read': { capacity: 20, refillPerSec: 5 },
   199→   199→  'conversation:join': { capacity: 30, refillPerSec: 10 },
   200→   200→}
   201→   201→
   202→   202→/**
   203→   203→ * Force-disconnect every socket owned by `userId`. Call this when the
   204→   204→ * user's tokenVersion is bumped (password change, role demotion, remote
   205→   205→ * logout) so that they have to re-authenticate to keep using real-time
   206→   206→ * channels.
   207→   207→ *
   208→   208→ * IMPORTANT: We do NOT clear `userSockets.get(userId)` here. The disconnect
   209→   209→ * is async — the socket's `disconnect` event fires later, and the
   210→   210→ * `disconnect` handler (below) is what cleans up `sockets.delete(socket.id)`
   211→   211→ * + calls `setUserOffline` when the set becomes empty. If we cleared here,
   212→   212→ * the disconnect handler would find an empty Set / undefined map and skip
   213→   213→ * the offline transition — leaving isOnline=true in the DB forever.
   214→   214→ */
   215→   215→export function kickUserSockets(userId: string) {
   216→   216→  const sockets = userSockets.get(userId)
   217→   217→  if (!sockets) return
   218→   218→  // Snapshot to array so we don't mutate during iteration.
   219→   219→  const snapshot = Array.from(sockets)
   220→   220→  for (const socketId of snapshot) {
   221→   221→    const socket = ioInstance?.sockets.sockets.get(socketId)
   222→   222→    if (socket) socket.disconnect(true)
   223→   223→  }
   224→   224→}

   226→   226→function rateLimit(socket: Socket, event: string): boolean {
   227→   227→  // H4 fix: FAIL-CLOSED для unmapped events. Раньше возвращалось `true`
   228→   228→  // (разрешить), что означало: любое новое событие, добавленное без записи в
   229→   229→  // RATE_LIMITS, полностью обходило лимитирование (DoS-поверхность). Теперь
   230→   230→  // применяем консервативный default-лимит (1 запрос в секунду, burst 3) —
   231→   231→  // достаточно для легитимных редких событий, но блокирует flood.
   232→   232→  const cfg = RATE_LIMITS[event] ?? { capacity: 3, refillPerSec: 1 }
   233→   233→  const map = (socket.data.rateBuckets ??= new Map<string, RateBucket>()) as Map<string, RateBucket>
   234→   234→  let bucket = map.get(event)
   235→   235→  const now = Date.now()
   236→   236→  if (!bucket) {
   237→   237→    bucket = { tokens: cfg.capacity, lastRefill: now }
   238→   238→    map.set(event, bucket)
   239→   239→  } else {
   240→   240→    const elapsed = (now - bucket.lastRefill) / 1000
   241→   241→    bucket.tokens = Math.min(cfg.capacity, bucket.tokens + elapsed * cfg.refillPerSec)
   242→   242→    bucket.lastRefill = now
   243→   243→  }
   244→   244→  if (bucket.tokens < 1) return false
   245→   245→  bucket.tokens -= 1
   246→   246→  return true
   247→   247→}

   303→   303→export function registerChatHandlers(io: IoServer) {
   304→   304→  ioInstance = io
   305→   305→  io.use(async (socket, next) => {
   306→   306→    const auth = await tryAuth(socket)
   307→   307→    if (!auth) return next(new Error('Unauthorized'))
   308→   308→    ;(socket.data as any).auth = auth
   309→   309→    next()
   310→   310→  })
   311→   311→
   312→   312→  io.on('connection', async (socket: Socket) => {
   313→   313→    const auth = (socket.data as any).auth as { userId: string; username: string }
   314→   314→    if (!auth) {
   315→   315→      socket.disconnect(true)
   316→   316→      return
   317→   317→    }
   318→   318→
   319→   319→    const { userId, username } = auth
   320→   320→
   321→   321→    // SECURITY: cap concurrent sockets per user to prevent DoS. A malicious
   322→   322→    // or buggy client can otherwise open thousands of sockets per second,
   323→   323→    // each triggering setUserOnline + presence broadcast + auto-join loop.
   324→   324→    // 10 is enough for any legitimate use (multiple tabs + multiple devices).
   325→   325→    const MAX_SOCKETS_PER_USER = 10
   326→   326→    const existingSockets = userSockets.get(userId)
   327→   327→    if (existingSockets && existingSockets.size >= MAX_SOCKETS_PER_USER) {
   328→   328→      console.warn(`[SOCKET] Rejecting socket for ${username}: ${existingSockets.size} sockets already open (cap=${MAX_SOCKETS_PER_USER})`)
   329→   329→      socket.emit('error', { message: 'Too many concurrent connections' })
   330→   330→      socket.disconnect(true)
   331→   331→      return
   332→   332→    }
   333→   333→
   334→   334→    socketToUser.set(socket.id, { userId, username })
   335→   335→    let sockets = userSockets.get(userId)
   336→   336→    if (!sockets) {
   337→   337→      sockets = new Set()
   338→   338→      userSockets.set(userId, sockets)
   339→   339→    }
   340→   340→    sockets.add(socket.id)
   341→   341→    void setUserOnline(userId)
   342→   342→    void broadcastPresenceToPartners(io, userId, 'user:online', { userId, username })
   343→   343→
   344→   344→    // ========================================================================
   345→   345→    // PERSONAL + ADMIN ROOMS
   346→   346→    // Every socket joins `user:<userId>` so we can target events (like
   347→   347→    // lead:status-changed) directly at the user regardless of which
   348→   348→    // conversation they're viewing. If the user is an admin, they also
   349→   349→    // join the `admins` room — used for admin-wide broadcasts such as
   350→   350→    // lead status updates visible to all admin studios.
   351→   351→    // ========================================================================
   352→   352→    socket.join(`user:${userId}`)
   353→   353→    try {
   354→   354→      const u = await prisma.user.findUnique({
   355→   355→        where: { id: userId },
   356→   356→        select: { role: true },
   357→   357→      })
   358→   358→      if (u?.role === 'admin') {
   359→   359→        socket.join('admins')
   360→   360→      }
   361→   361→    } catch {
   362→   362→      /* non-critical */
   363→   363→    }
   364→   364→
   365→   365→    // ========================================================================
   366→   366→    // AUTO-JOIN all of the user's conversations.
   367→   367→    // This is the critical fix: previously the frontend only joined a room
   368→   368→    // when the user opened a specific conversation. That meant messages in
   369→   369→    // other conversations were NEVER delivered to the user when they were
   370→   370→    // not viewing that chat (or were elsewhere in the app). Now the server
   371→   371→    // joins every connected socket to every conversation the user is a
   372→   372→    // participant of, so `message:received` broadcasts reach them no matter
   373→   373→    // which view they're on. The client-side `conversation:join` is kept as
   374→   374→    // a no-op compatibility shim.
   375→   375→    //
   376→   376→    // v13.1 (audit P1-12 fix): emit a 'ready' event after the auto-join
   377→   377→    // completes so the client can wait for it before sending messages.
   378→   378→    // Previously the IIFE was fire-and-forget — between socket connection
   379→   379→    // and the IIFE completing, any `message:received` broadcast to a
   380→   380→    // conversation room would NOT reach this socket. The client doesn't
   381→   381→    // strictly need to wait on 'ready' (older clients still work), but
   382→   382→    // new clients CAN wait to guarantee delivery.
   383→   383→    ;(async () => {
   384→   384→      try {
   385→   385→        const participations = await prisma.conversationParticipant.findMany({
   386→   386→          where: { userId },
   387→   387→          select: { conversationId: true },
   388→   388→        })
   389→   389→        for (const p of participations) {
   390→   390→          socket.join(`conversation:${p.conversationId}`)
   391→   391→        }
   392→   392→        socket.data.ready = true
   393→   393→        socket.emit('ready', { userId })
   394→   394→      } catch (e) {
   395→   395→        console.error('[SOCKET] auto-join conversations error:', e)
   396→   396→      }
   397→   397→    })()
   398→   398→
   399→   399→    // ====== Conversation room management (kept for backwards compat) ======
   400→   400→    socket.on('conversation:join', async (conversationId: string) => {
   401→   401→      try {
   402→   402→        if (!conversationId || typeof conversationId !== 'string') return
   403→   403→        const ok = await isParticipant(conversationId, userId)
   404→   404→        if (!ok) {
   405→   405→          socket.emit('error', { event: 'conversation:join', error: 'Not a participant' })
   406→   406→          return
   407→   407→        }
   408→   408→        socket.join(`conversation:${conversationId}`)
   409→   409→      } catch (e) {
   410→   410→        socket.emit('error', { event: 'conversation:join', error: 'Internal error' })
   411→   411→      }
   412→   412→    })
   413→   413→
   414→   414→    // conversation:leave is now a NO-OP — the server auto-joins every
   415→   415→    // socket to every conversation the user is a participant of. Leaving a
   416→   416→    // room would silently break `message:received` delivery for that
   417→   417→    // conversation, killing in-app toast notifications, unread-badge
   418→   418→    // increments, and chat-list preview refreshes for any conversation
   419→   419→    // the user has ever opened. Per-conversation filtering is done
   420→   420→    // client-side in chat.tsx.
   421→   421→    //
   422→   422→    // The handler is kept (not deleted) so legacy clients that still emit

   429→   429→    // Server-side TTL: if a client sends `typing:start` and then crashes /
   430→   430→    // closes the tab before sending `typing:stop`, other participants would
   431→   431→    // see a permanent "печатает…" indicator. We arm a 5-second timer on each
   432→   432→    // `typing:start` — if the client doesn't refresh or explicitly stop, we
   433→   433→    // auto-broadcast `typing:stop` to the room.
   434→   434→    // Map keyed by `${conversationId}:${userId}` so multiple conversations

   436→   436→    // collide. `timer.unref()` prevents the timer from keeping the event
   437→   437→    // loop alive on shutdown.
   438→   438→    const typingTimerKey = (conversationId: string) => `${conversationId}:${userId}`
   439→   439→    const clearTypingTimer = (conversationId: string) => {
   440→   440→      const t = typingTimers.get(typingTimerKey(conversationId))
   441→   441→      if (t) {
   442→   442→        clearTimeout(t)
   443→   443→        typingTimers.delete(typingTimerKey(conversationId))
   444→   444→      }
   445→   445→    }

   452→   452→        socket.to(`conversation:${conversationId}`).emit('typing:start', { conversationId, userId, username })
   453→   453→

   455→   455→        // (client refreshed `typing:start`), clear it first to extend the
   456→   456→        // window.

   861→   861→
   862→   862→          // Notify the recipient (across all their sockets — multi-tab)
   863→   863→          emitToUserSockets(io, payload.recipientId, 'call:incoming', {
   864→   864→            callId: call.id,
   865→   865→            conversationId: payload.conversationId,
   866→   866→            type: payload.type,
   867→   867→            caller: caller
   868→   868→              ? {
   869→   869→                  id: caller.id,
   870→   870→                  username: caller.username,
   871→   871→                  displayName: caller.displayName,
   872→   872→                  avatar: caller.avatar,
   873→   873→                }
   874→   874→              : null,
   875→   875→          })
   876→   876→          socket.emit('call:started', { callId: call.id, conversationId: payload.conversationId })
   877→   877→
   878→   878→          // ====== MISSED CALL TIMEOUT ======
   879→   879→          // If the recipient doesn't answer within 45 seconds, auto-mark the
   880→   880→          // call as 'missed' and notify both parties. Without this, the call
   881→   881→          // stays in 'ringing' forever — the caller sees "Вызов…" endlessly
   882→   882→          // and the recipient's phone keeps vibrating if push was sent.
   883→   883→          //
   884→   884→          // 45s is the industry standard (Telegram, WhatsApp, Viber all use
   885→   885→          // 30-60s). Long enough for the recipient to grab their phone, short
   886→   886→          // enough to not waste the caller's time.
   887→   887→          setTimeout(() => {
   888→   888→            (async () => {
   889→   889→              try {
   890→   890→                const fresh = await prisma.call.findUnique({
   891→   891→                  where: { id: call.id },
   892→   892→                  select: { id: true, status: true, callerId: true, recipientId: true },
   893→   893→                })
   894→   894→                if (!fresh) return
   895→   895→                if (fresh.status !== 'ringing') return // already accepted/rejected/cancelled
   896→   896→                await prisma.call.update({
   897→   897→                  where: { id: call.id },
   898→   898→                  data: { status: 'missed', endedAt: new Date() },
   899→   899→                })
   900→   900→                // Notify both: caller sees "missed", recipient sees "missed" too
   901→   901→                emitToUserSockets(io, fresh.callerId, 'call:ended', {
   902→   902→                  callId: call.id,
   903→   903→                  reason: 'missed',
   904→   904→                  duration: null,
   905→   905→                })
   906→   906→                emitToUserSockets(io, fresh.recipientId, 'call:ended', {
   907→   907→                  callId: call.id,
   908→   908→                  reason: 'missed',
   909→   909→                  duration: null,
   910→   910→                })
   911→   911→              } catch (e) {
   912→   912→                console.error('[CALL] missed timeout error:', e instanceof Error ? e.message : e)
   913→   913→              }
   914→   914→            })()
   915→   915→          }, 45_000).unref?.() // unref so the timer doesn't keep Node alive on shutdown
   916→   916→
   917→   917→          // WEB PUSH for incoming call — fires when the recipient has NO
   918→   918→          // active socket connection (app fully closed or backgrounded).
   919→   919→          // The push notification includes Answer/Decline action buttons
   920→   920→          // that the SW handles via `notificationclick` -> event.action.
   921→   921→          // requireInteraction: true so the notification persists until
   922→   922→          // the user acts on it (otherwise Android auto-dismisses after
   923→   923→          // ~5s and the user misses the call).
   924→   924→          if (!isUserOnline(payload.recipientId)) {
   925→   925→            const callerName = caller?.displayName || caller?.username || 'Пользователь'
   926→   926→            const callTitle = `Входящий ${payload.type === 'video' ? 'видеозвонок' : 'аудиозвонок'}`
   927→   927→            void sendPushToUser(payload.recipientId, {
   928→   928→              title: callTitle,
   929→   929→              body: `От ${callerName}`,
   930→   930→              tag: `call-${call.id}`,
   931→   931→              url: '/?view=chat',
   932→   932→              data: {
   933→   933→                conversationId: payload.conversationId,
   934→   934→                callId: call.id,
   935→   935→                callType: payload.type,
   936→   936→                callerId: userId,
   937→   937→              },
   938→   938→              requireInteraction: true,
   939→   939→              renotify: false, // call tag is already unique per call
   940→   940→              actions: [
   941→   941→                { action: 'answer', title: 'Ответить' },
   942→   942→                { action: 'decline', title: 'Отклонить' },
   943→   943→              ],
   944→   944→            })
   945→   945→          }
   946→   946→        } catch (e: any) {
   947→   947→          socket.emit('error', { event: 'call:start', error: e?.message || 'Call start failed' })
   948→   948→        }
   949→   949→      },
   950→   950→    )
   951→   951→
   952→   952→    // Accept an incoming call
   953→   953→    socket.on('call:accept', async (payload: { callId: string }) => {
   954→   954→      try {
   955→   955→        if (!payload?.callId) return
   956→   956→        const call = await prisma.call.findUnique({
   957→   957→          where: { id: payload.callId },
   958→   958→          select: { id: true, callerId: true, recipientId: true, status: true, conversationId: true },
   959→   959→        })
   960→   960→        if (!call) {
   961→   961→          socket.emit('error', { event: 'call:accept', error: 'Call not found' })
   962→   962→          return
   963→   963→        }
   964→   964→        if (call.recipientId !== userId) {
   965→   965→          socket.emit('error', { event: 'call:accept', error: 'Not the recipient' })
   966→   966→          return
   967→   967→        }
   968→   968→        if (call.status !== 'ringing') {
   969→   969→          socket.emit('error', { event: 'call:accept', error: `Call already ${call.status}` })
   970→   970→          return
   971→   971→        }
   972→   972→        await prisma.call.update({
   973→   973→          where: { id: call.id },
   974→   974→          data: { status: 'accepted', startedAt: new Date() },
   975→   975→        })
   976→   976→        emitToUserSockets(io, call.callerId, 'call:accepted', { callId: call.id })
   977→   977→      } catch (e: any) {
   978→   978→        socket.emit('error', { event: 'call:accept', error: e?.message })
   979→   979→      }
   980→   980→    })
   981→   981→
   982→   982→    // Reject an incoming call
   983→   983→    socket.on('call:reject', async (payload: { callId: string }) => {
   984→   984→      try {
   985→   985→        if (!payload?.callId) return
   986→   986→        const call = await prisma.call.findUnique({
   987→   987→          where: { id: payload.callId },
   988→   988→          select: { id: true, callerId: true, recipientId: true, status: true },
   989→   989→        })
   990→   990→        if (!call) return
   991→   991→        if (call.recipientId !== userId && call.callerId !== userId) return
   992→   992→        if (call.status !== 'ringing') return
   993→   993→        await prisma.call.update({
   994→   994→          where: { id: call.id },
   995→   995→          data: { status: 'rejected', endedAt: new Date() },
   996→   996→        })
   997→   997→        emitToUserSockets(io, call.callerId, 'call:rejected', { callId: call.id })
   998→   998→        emitToUserSockets(io, call.recipientId, 'call:rejected', { callId: call.id })
   999→   999→      } catch (e: any) {
  1000→  1000→        socket.emit('error', { event: 'call:reject', error: e?.message })
  1001→  1001→      }
  1002→  1002→    })
  1003→  1003→
  1004→  1004→    // End an active call
  1005→  1005→    socket.on('call:end', async (payload: { callId: string; reason?: string }) => {
  1006→  1006→      try {
  1007→  1007→        if (!payload?.callId) return
  1008→  1008→        const call = await prisma.call.findUnique({
  1009→  1009→          where: { id: payload.callId },
  1010→  1010→          select: { id: true, callerId: true, recipientId: true, status: true, startedAt: true },
  1011→  1011→        })
  1012→  1012→        if (!call) return
  1013→  1013→        if (call.callerId !== userId && call.recipientId !== userId) return
  1014→  1014→
  1015→  1015→        const duration = call.startedAt
  1016→  1016→          ? Math.floor((Date.now() - call.startedAt.getTime()) / 1000)
  1017→  1017→          : null
  1018→  1018→        await prisma.call.update({
  1019→  1019→          where: { id: call.id },
  1020→  1020→          data: { status: 'ended', endedAt: new Date(), duration },
  1021→  1021→        })
  1022→  1022→        emitToUserSockets(io, call.callerId, 'call:ended', { callId: call.id, reason: payload.reason || 'ended', duration })
  1023→  1023→        emitToUserSockets(io, call.recipientId, 'call:ended', { callId: call.id, reason: payload.reason || 'ended', duration })
  1024→  1024→      } catch (e: any) {
  1025→  1025→        socket.emit('error', { event: 'call:end', error: e?.message })
  1026→  1026→      }
  1027→  1027→    })
  1028→  1028→
  1029→  1029→    // Cancel an outgoing call before the recipient answers
  1030→  1030→    socket.on('call:cancel', async (payload: { callId: string }) => {
  1031→  1031→      try {
  1032→  1032→        if (!payload?.callId) return
  1033→  1033→        const call = await prisma.call.findUnique({
  1034→  1034→          where: { id: payload.callId },
  1035→  1035→          select: { id: true, callerId: true, recipientId: true, status: true },
  1036→  1036→        })
  1037→  1037→        if (!call) return
  1038→  1038→        if (call.callerId !== userId) return
  1039→  1039→        if (call.status !== 'ringing') return
  1040→  1040→        await prisma.call.update({
  1041→  1041→          where: { id: call.id },
  1042→  1042→          data: { status: 'cancelled', endedAt: new Date() },
  1043→  1043→        })
  1044→  1044→        emitToUserSockets(io, call.recipientId, 'call:cancelled', { callId: call.id })
  1045→  1045→      } catch (e: any) {
  1046→  1046→        socket.emit('error', { event: 'call:cancel', error: e?.message })
  1047→  1047→      }
  1048→  1048→    })
  1049→  1049→
  1050→  1050→    // WebRTC signaling — relay offer/answer/ICE candidates to the peer.
  1051→  1051→    // The server does NOT touch the media; it just forwards signaling messages.
  1052→  1052→    socket.on('call:signal', async (payload: { callId: string; to: string; data: any }) => {
  1053→  1053→      try {
  1054→  1054→        if (!payload?.callId || !payload?.to || !payload?.data) return
  1055→  1055→        // B-HIGH-007: cap relayed signal payload to 16 KiB. Without this, a
  1056→  1056→        // malicious or buggy client can ship an arbitrarily-large `data`
  1057→  1057→        // object (e.g. SDP with embedded data channels, base64 media) which
  1058→  1058→        // the server happily re-serialises and broadcasts to every socket of
  1059→  1059→        // the peer — a trivial amplification / memory-exhaustion vector.
  1060→  1060→        // 16 KiB is far above any legit SDP+ICE bundle (typical: 2–4 KiB).
  1061→  1061→        const dataStr = JSON.stringify(payload.data)
  1062→  1062→        if (dataStr.length > 16_384) {
  1063→  1063→          socket.emit('error', { event: 'call:signal', error: 'Signal payload too large' })
  1064→  1064→          return
  1065→  1065→        }
  1066→  1066→        // Verify the caller is part of this call
  1067→  1067→        const call = await prisma.call.findUnique({
  1068→  1068→          where: { id: payload.callId },
  1069→  1069→          select: { callerId: true, recipientId: true, status: true },
  1070→  1070→        })
  1071→  1071→        if (!call) return
  1072→  1072→        if (call.callerId !== userId && call.recipientId !== userId) return
  1073→  1073→        // v13.0 (audit P0-3 fix): the previous version only validated that
  1074→  1074→        // the SENDER (userId) is a participant, but did not validate that
  1075→  1075→        // payload.to (the recipient of the relayed signaling) is the OTHER
  1076→  1076→        // participant. An attacker in a call with user B could relay SDP/ICE
  1077→  1077→        // to ANY arbitrary user ID — leaking their ICE candidates (IPs)
  1078→  1078→        // and spamming their UI. Now strictly require payload.to to be the
  1079→  1079→        // other party of the same call.
  1080→  1080→        if (payload.to !== call.callerId && payload.to !== call.recipientId) return
  1081→  1081→        if (payload.to === userId) return // cannot signal yourself
  1082→  1082→        // v13.0 (audit P1-11 fix): only relay signaling once the call is
  1083→  1083→        // accepted. Previously `ringing` was allowed — letting the caller
  1084→  1084→        // push SDP/ICE to the recipient before they consented to the call,
  1085→  1085→        // leaking their ICE candidates. Now strictly require `accepted`.
  1086→  1086→        if (call.status !== 'accepted') return
  1087→  1087→        // Forward to the peer's sockets
  1088→  1088→        emitToUserSockets(io, payload.to, 'call:signal', {
  1089→  1089→          callId: payload.callId,
  1090→  1090→          from: userId,
  1091→  1091→          data: payload.data,
  1092→  1092→        })
  1093→  1093→      } catch (e: any) {
  1094→  1094→        socket.emit('error', { event: 'call:signal', error: e?.message })
  1095→  1095→      }
  1096→  1096→    })
  1097→  1097→
  1098→  1098→    // ====== Disconnect ======
  1099→  1099→    socket.on('disconnect', async () => {
  1100→  1100→      socketToUser.delete(socket.id)
  1101→  1101→      const sockets = userSockets.get(userId)
  1102→  1102→      if (sockets) {
  1103→  1103→        sockets.delete(socket.id)
  1104→  1104→        if (sockets.size === 0) {
  1105→  1105→          userSockets.delete(userId)
  1106→  1106→          await setUserOffline(userId)
  1107→  1107→          void broadcastPresenceToPartners(io, userId, 'user:offline', { userId, username })
  1108→  1108→
  1109→  1109→          // ====== AUTO-END ACTIVE CALLS ON DISCONNECT ======
  1110→  1110→          // If the user had an active call and their socket dropped without
  1111→  1111→          // calling call:end, we need to clean up. Otherwise the OTHER party
  1112→  1112→          // stays in the call forever waiting for a peer that's gone.
  1113→  1113→          //
  1114→  1114→          // We check 3 seconds after disconnect — this gives a brief window
  1115→  1115→          // for the client to reconnect (mobile network flaps) before we
  1116→  1116→          // declare the call dead. If the user reconnects within 3s, their
  1117→  1117→          // new socket re-registers and the call continues normally.
  1118→  1118→          setTimeout(() => {
  1119→  1119→            (async () => {
  1120→  1120→              try {
  1121→  1121→                // If the user reconnected (any socket is alive), skip cleanup.
  1122→  1122→                if (isUserOnline(userId)) return
  1123→  1123→
  1124→  1124→                // Find any active call where this user is caller OR recipient
  1125→  1125→                // AND the call is in an active state (accepted/connected).
  1126→  1126→                const activeCalls = await prisma.call.findMany({
  1127→  1127→                  where: {
  1128→  1128→                    OR: [{ callerId: userId }, { recipientId: userId }],
  1129→  1129→                    status: { in: ['accepted', 'ringing'] },
  1130→  1130→                  },
  1131→  1131→                  select: {
  1132→  1132→                    id: true,
  1133→  1133→                    callerId: true,
  1134→  1134→                    recipientId: true,
  1135→  1135→                    status: true,
  1136→  1136→                    startedAt: true,
  1137→  1137→                    conversationId: true,
  1138→  1138→                  },
  1139→  1139→                })
  1140→  1140→                for (const c of activeCalls) {
  1141→  1141→                  const duration = c.startedAt
  1142→  1142→                    ? Math.floor((Date.now() - c.startedAt.getTime()) / 1000)
  1143→  1143→                    : null
  1144→  1144→                  // ringing → missed (caller's perspective) or cancelled (recipient's)
  1145→  1145→                  // accepted → ended (with duration)
  1146→  1146→                  const newStatus = c.status === 'ringing'
  1147→  1147→                    ? (c.callerId === userId ? 'cancelled' : 'missed')
  1148→  1148→                    : 'ended'
  1149→  1149→                  await prisma.call.update({
  1150→  1150→                    where: { id: c.id },
  1151→  1151→                    data: { status: newStatus, endedAt: new Date(), duration },
  1152→  1152→                  })
  1153→  1153→                  const otherId = c.callerId === userId ? c.recipientId : c.callerId
  1154→  1154→                  emitToUserSockets(io, otherId, 'call:ended', {
  1155→  1155→                    callId: c.id,
  1156→  1156→                    reason: newStatus === 'missed' ? 'missed' : newStatus === 'cancelled' ? 'cancelled' : 'peer-disconnected',
  1157→  1157→                    duration,
  1158→  1158→                  })
  1159→  1159→                  // Also notify the disconnecting user (in case they reconnect
  1160→  1160→                  // — they should see the call ended)
  1161→  1161→                  emitToUserSockets(io, userId, 'call:ended', {
  1162→  1162→                    callId: c.id,
  1163→  1163→                    reason: newStatus === 'missed' ? 'missed' : newStatus === 'cancelled' ? 'cancelled' : 'peer-disconnected',
  1164→  1164→                    duration,
  1165→  1165→                  })
  1166→  1166→                }
  1167→  1167→              } catch (e) {
  1168→  1168→                console.error('[CALL] disconnect cleanup error:', e instanceof Error ? e.message : e)
  1169→  1169→              }
  1170→  1170→            })()
  1171→  1171→          }, 3_000).unref?.() // 3s grace period for reconnect
  1172→  1172→        }
  1173→  1173→      }
  1174→  1174→    })
  1175→  1175→  })
  1176→  1176→}
