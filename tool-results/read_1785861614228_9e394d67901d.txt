   115→function pickStatusGroup(message: string): 'db' | 'calc' | 'thinking' {
   116→  const m = (message || '').toLowerCase()
   117→  // Калькуляция — цены, стоимость, расчёт
   118→  if (/цен|стоим|расчёт|расчет|сколько стоит|подсчитай|считай|во сколько|за сколько/.test(m)) {
   119→    return 'calc'
   120→  }
   121→  // БД — товары, категории, stories, баннеры, заказы, пользователи
   122→  if (/товар|катег|stor|баннер|заказ|польз|клиент|сколько у нас|покажи|найди/.test(m)) {
   123→    return 'db'
   124→  }
   125→  return 'thinking'
   126→}

   131→function useVoiceInput(opts: {
   132→  onResult: (text: string) => void
   133→  onInterim?: (text: string) => void
   134→  onStart?: () => void
   135→  onEnd?: () => void
   136→  lang?: string
   137→  continuous?: boolean
   138→}) {
   139→  const [listening, setListening] = useState(false)
   140→  const recRef = useRef<any>(null)
   141→  const startingRef = useRef(false)
   142→  // v5: фиксируем onEnd, чтобы вызывать его когда recognition завершился сам
   143→  // (например, по таймауту браузера). Это нужно для логики «Вы закончили?».
   144→  const onEndRef = useRef<() => void>(() => {})
   145→  onEndRef.current = opts.onEnd || (() => {})
   146→
   147→  const stop = useCallback(() => {
   148→    startingRef.current = false
   149→    if (recRef.current) {
   150→      try {
   151→        recRef.current.onresult = null
   152→        recRef.current.onerror = null
   153→        recRef.current.onend = null
   154→        recRef.current.onstart = null
   155→        recRef.current.abort?.()
   156→        recRef.current.stop?.()
   157→      } catch {}
   158→      recRef.current = null
   159→    }
   160→    setListening(false)
   161→  }, [])
   162→
   163→  const start = useCallback(() => {
   164→    if (typeof window === 'undefined') return
   165→    if (listening || startingRef.current) return
   166→    if (recRef.current) return
   167→    const SR = (window as any).SpeechRecognition || (window as any).webkitSpeechRecognition
   168→    if (!SR) {
   169→      alert('Голосовой ввод не поддерживается. Используйте Chrome, Edge или Safari.')
   170→      return
   171→    }
   172→    startingRef.current = true
   173→    const rec = new SR()
   174→    rec.lang = opts.lang || 'ru-RU'
   175→    // v5: continuous = true — чтобы recognition не закрывался после каждой фразы.
   176→    // Это позволяет нам самим управлять моментом «пользователь замолчал».
   177→    rec.continuous = opts.continuous !== false
   178→    rec.interimResults = true
   179→    rec.maxAlternatives = 1
   180→    rec.onstart = () => {
   181→      startingRef.current = false
   182→      setListening(true)
   183→      opts.onStart?.()
   184→    }
   185→    rec.onerror = (e: any) => {
   186→      startingRef.current = false
   187→      setListening(false)
   188→      console.warn('[VoiceInput] error:', e?.error)
   189→    }
   190→    rec.onend = () => {
   191→      startingRef.current = false
   192→      setListening(false)
   193→      // v24.3 BUGFIX: clear recRef.current so subsequent start() calls can
   194→      // instantiate a fresh SpeechRecognition. Previously, on desktop Chrome,
   195→      // recognition.onend fired after each utterance/silence but recRef.current
   196→      // was never cleared — so the guard `if (recRef.current) return` in
   197→      // start() blocked the next start() call, and follow-up questions were
   198→      // never captured. Mobile Chrome with continuous=true keeps recognition
   199→      // alive across utterances (onend doesn't fire mid-conversation), which
   200→      // is why mobile worked but desktop didn't.
   201→      recRef.current = null
   202→      // v5: браузер сам завершил recognition (timeout или no-speech).
   203→      // Уведомляем caller, чтобы он мог решить, перезапускать или нет.
   204→      onEndRef.current?.()
   205→    }
   206→    rec.onresult = (e: any) => {
   207→      let finalText = ''
   208→      let interimText = ''
   209→      for (let i = e.resultIndex; i < e.results.length; i++) {
   210→        const r = e.results[i]
   211→        if (r.isFinal) finalText += r[0].transcript
   212→        else interimText += r[0].transcript
   213→      }
   214→      if (interimText) opts.onInterim?.(interimText)
   215→      if (finalText) {
   216→        const cleaned = finalText.trim()
   217→        if (cleaned) opts.onResult(cleaned)
   218→      }
   219→    }
   220→    recRef.current = rec
   221→    try {
   222→      rec.start()
   223→    } catch (err) {
   224→      startingRef.current = false
   225→      console.warn('[VoiceInput] start() threw:', err)
   226→    }
   227→  }, [opts, listening, stop])
   228→
   229→  useEffect(() => () => stop(), [stop])
   230→  return { listening, start, stop }
   231→}

   233→// ---------------------------------------------------------------------------
   234→//  Voice output — speechSynthesis TTS (по умолчанию выключен, но доступен)
   235→// ---------------------------------------------------------------------------
   236→function useVoiceOutput() {
   237→  const [enabled, setEnabled] = useState(false)
   238→  const [speaking, setSpeaking] = useState(false)
   239→  const voiceRef = useRef<SpeechSynthesisVoice | null>(null)
   240→
   241→  useEffect(() => {
   242→    if (typeof window === 'undefined' || !('speechSynthesis' in window)) return
   243→    try { window.speechSynthesis.cancel() } catch {}
   244→    return () => {
   245→      try { window.speechSynthesis.cancel() } catch {}
   246→    }
   247→  }, [])
   248→
   249→  const speak = useCallback((_text: string, onEnd?: () => void) => {
   250→    // v5: TTS по умолчанию выключен (как в v22). Сразу вызываем onEnd,
   251→    // чтобы цикл продолжался без роботизированного голоса.
   252→    onEnd?.()
   253→  }, [])
   254→
   255→  const stop = useCallback(() => {
   256→    if (typeof window === 'undefined' || !('speechSynthesis' in window)) return
   257→    try { window.speechSynthesis.cancel() } catch {}
   258→    setSpeaking(false)
   259→  }, [])
   260→
   261→  return { enabled, setEnabled, speak, stop, speaking }
   262→}

   987→function ResponseCards({
   988→  cards,
   989→  onOpenProduct,
   990→}: {
   991→  cards: ChatCard[]
   992→  onOpenProduct?: (id: string) => void
   993→}) {
   994→  // Группируем product/similar_products карточки по category
   995→  // Если у карточки есть data.category — берём его; иначе — общий "Товары"
   996→  type Group = { title: string; emoji: string; products: any[] }
   997→  const productCards = cards.filter(c => c.kind === 'product' || c.kind === 'similar_products')
   998→  const otherCards = cards.filter(c => c.kind !== 'product' && c.kind !== 'similar_products')
   999→
  1000→  const groups: Group[] = []
  1001→  for (const card of productCards) {
  1002→    const items = Array.isArray(card.data) ? card.data : [card.data]
  1003→    for (const item of items) {
  1004→      const cat = item?.category || item?.categoryName || (card.kind === 'similar_products' ? 'Похожие товары' : 'Товары')

  1006→      if (!g) {
  1007→        const emoji = categoryEmoji(cat)
  1008→        g = { title: cat, emoji, products: [] }
  1009→        groups.push(g)
  1010→      }
  1011→      g.products.push(item)
  1012→    }

  1014→
  1015→  return (

  1029→                key={i}

  1046→function categoryEmoji(cat: string): string {
  1047→  const c = (cat || '').toLowerCase()
  1048→  if (c.includes('подар')) return '🎁'
  1049→  if (c.includes('баннер') || c.includes('banner')) return '🖨'
  1050→  if (c.includes('печат') || c.includes('print')) return '🖨'
  1051→  if (c.includes('визитк')) return '💳'
  1052→  if (c.includes('футбол') || c.includes('одежд')) return '👕'
  1053→  if (c.includes('кружк')) return '☕'
  1054→  if (c.includes('плакат') || c.includes('постер')) return '🖼'
  1055→  if (c.includes('буклет') || c.includes('листовк')) return '📄'
  1056→  if (c.includes('наклейк') || c.includes('стикер')) return '✨'
  1057→  if (c.includes('книг') || c.includes('журнал')) return '📚'
  1058→  if (c.includes('упаков')) return '📦'
  1059→  if (c.includes('сувенир')) return '🎀'
  1060→  return '🛍'
  1061→}

  1067→function CompactProductCard({ product, onClick }: { product: any; onClick?: () => void }) {
  1068→  if (!product) return null
  1069→  const name = product.name || product.title || 'Товар'
  1070→  const price = product.price != null ? product.price : product.basePrice
  1071→  const oldPrice = product.oldPrice
  1072→  const image = product.image || product.imageUrl || product.coverImage || (Array.isArray(product.images) ? product.images[0] : null)
  1073→  const currencySymbol = (() => {
  1074→    const c = (product.currency || 'RUB').toString().toUpperCase()
  1075→    if (c === 'RUB' || c === 'РУБ') return '₽'
  1076→    if (c === 'USD') return '$'
  1077→    if (c === 'EUR') return '€'
  1078→    if (c === 'KZT') return '₸'
  1079→    if (c === 'UAH') return '₴'
  1080→    if (c === 'BYN') return 'Br'
  1081→    return c
  1082→  })()
  1083→  const isAction = product.isAction || product.discount
  1084→  const isNew = product.isNew
  1085→  const isPopular = product.isPopular
  1086→  const inStock = product.inStock !== false && product.stockStatus !== 'out_of_stock' && product.quantity !== 0
  1087→  const discountPct = oldPrice && price != null && oldPrice > price
  1088→    ? Math.round((1 - price / oldPrice) * 100)
  1089→    : null
  1090→
  1091→  return (
  1092→    <button
  1093→      onClick={onClick}
  1094→      className="snap-start shrink-0 text-left w-[160px] rounded-xl overflow-hidden bg-white/5 border border-white/10 hover:border-violet-400/40 hover:bg-white/10 transition-all group"

  1099→          <img
  1100→            src={typeof image === 'string' ? image : image.url}
  1101→            alt={name}
  1102→            className="absolute inset-0 w-full h-full object-cover group-hover:scale-105 transition-transform"
  1103→            loading="lazy"
  1104→          />
  1105→        ) : (
  1106→          <div className="absolute inset-0 grid place-items-center">
  1107→            <Package className="h-5 w-5 text-white/30" />
  1108→          </div>
  1109→        )}
  1110→        {/* Discount / Action badge */}
  1111→        {discountPct ? (
  1112→          <div className="absolute top-1 left-1 px-1.5 py-0.5 rounded-md bg-rose-500/90 text-white text-[9px] font-bold leading-none">
  1113→            -{discountPct}%
  1114→          </div>
  1115→        ) : isAction ? (
  1116→          <div className="absolute top-1 left-1 px-1.5 py-0.5 rounded-md bg-rose-500/90 text-white text-[9px] font-bold leading-none uppercase">
  1117→            Акция
  1118→          </div>
  1119→        ) : isNew ? (
  1120→          <div className="absolute top-1 left-1 px-1.5 py-0.5 rounded-md bg-emerald-500/90 text-white text-[9px] font-bold leading-none uppercase">
  1121→            Новинка
  1122→          </div>
  1123→        ) : isPopular ? (
  1124→          <div className="absolute top-1 left-1 px-1.5 py-0.5 rounded-md bg-amber-500/90 text-white text-[9px] font-bold leading-none uppercase">
  1125→            Хит
  1126→          </div>
  1127→        ) : null}
  1128→        {/* Stock indicator */}
  1129→        <div
  1130→          className={cn(
  1131→            'absolute top-1 right-1 h-1.5 w-1.5 rounded-full',
  1132→            inStock ? 'bg-emerald-400' : 'bg-rose-400'
  1133→          )}
  1134→          title={inStock ? 'В наличии' : 'Нет в наличии'}
  1135→        />
  1136→      </div>
  1137→      {/* Content — compact */}
  1138→      <div className="p-2 flex flex-col gap-0.5">
  1139→        <div className="text-[11px] text-white/90 font-medium line-clamp-2 leading-tight min-h-[26px]">
  1140→          {name}
  1141→        </div>
  1142→        <div className="flex items-baseline gap-1">
  1143→          {price != null && (
  1144→            <div className="text-xs font-bold text-violet-300">
  1145→              {Number(price).toLocaleString('ru-RU')} {currencySymbol}
  1146→            </div>
  1147→          )}
  1148→          {oldPrice && (
  1149→            <div className="text-[10px] text-white/40 line-through">
  1150→              {Number(oldPrice).toLocaleString('ru-RU')}
  1151→            </div>
  1152→          )}
  1153→        </div>
  1154→      </div>
  1155→    </button>
  1156→  )
