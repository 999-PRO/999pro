    59→function getInitialView(): View {
    60→  if (typeof window === 'undefined') return 'home'
    61→  const params = new URLSearchParams(window.location.search)
    62→  const v = params.get('view')
    63→  // v25.6: 'support' is kept as a valid view for backward-compat with deep
    64→  // links, but it now resolves to ContactsView (see routing below). Old
    65→  // bookmarks /?view=support still work — they show Contacts.
    66→  const validViews: View[] = ['home', 'catalog', 'club', 'chat', 'profile', 'search', 'studio', 'orders', 'reviews', 'support', 'contacts', 'settings', 'privacy', 'about', 'info', 'admin-login']
    67→  if (v && validViews.includes(v as View)) return v as View
    68→  return 'home'
    69→}

   639→function HomeView({
   640→  onNavigate,
   641→  onAuth,
   642→  onOpenProduct,
   643→  onStartConversation,
   644→}: {
   645→  onNavigate: (v: string) => void
   646→  onAuth: () => void
   647→  onOpenProduct: (id: string) => void
   648→  onStartConversation?: (userId: string) => void
   649→}) {
   650→  // v12.6.3: inline search state. When the user types a query, we hide
   651→  // Stories / Banners / SmartBlocks and show only the search results
   652→  // (which InlineSearch renders below itself, in normal flow). This
   653→  // matches the catalog UX exactly: type → results filter in place.
   654→  const [searchActive, setSearchActive] = useState(false)
   655→
   656→  // v16.5: Home page layout config — admin controls block order/visibility
   657→  // via Studio → "Главная". We only use isVisible() here; SmartBlocks already
   658→  // renders all 10 blocks internally, so we only gate the top-level sections.
   659→  const { isVisible } = useHomeLayout()
   660→
   661→  const handleQueryChange = useCallback((query: string, _hasResults: boolean) => {
   662→    // Search is "active" when there's a query with ≥2 chars (regardless of
   663→    // whether results have loaded yet). InlineSearch only shows results when
   664→    // hasQuery is true, so we mirror that here.
   665→    setSearchActive(query.trim().length >= 2)
   666→  }, [])
   667→
   668→  return (
   669→    <div className="pb-28 md:pb-0 page-top-padding">
   670→      {/* v18.4: Floating Live Info — "живая" информационная строка между
   671→          Header и Search. Текст парит в пространстве (без карточек/рамок),
   672→          ротация: погода → финансы → категории магазина → персональные.
   673→          Показывается только на главной странице. */}
   674→      <FloatingLiveInfo />
   675→
   676→      {/* v12.6.3: InlineSearch — reusable inline search, identical UX to the
   677→          catalog ("999 Store"). An inline <input> in normal flow; typing
   678→          filters results in place below the bar. NO overlay, NO modal, NO
   679→          page navigation. When active, the surrounding content (Stories,
   680→          Banners, SmartBlocks) is hidden so only results show — same as the
   681→          catalog. Sits directly below the Header, scrolls with the page. */}
   682→      <InlineSearch
   683→        onOpenProduct={onOpenProduct}
   684→        onStartConversation={onStartConversation}
   685→        onQueryChange={handleQueryChange}
   686→      />
   687→
   688→      {/* When search is NOT active, show the normal Home content.
   689→          When search IS active (query ≥ 2 chars), hide everything except
   690→          the search results (which InlineSearch renders itself). */}
   691→      {!searchActive && (
   692→        <>
   693→          {/* Hero header (desktop) — managed via Studio → "Hero блок" */}
   694→          {isVisible('hero') && <Hero onNavigate={onNavigate} />}
   695→
   696→          {/* v12.3.1: Stories component RESTORED (standalone module — not Feed). */}
   697→          {isVisible('stories') && <Stories />}
   698→
   699→          {/* Promo banner */}
   700→          {isVisible('banner') && <PromoBannerCarousel />}
   701→
   702→          {/* Smart blocks — single API call returns 10 personalized blocks.
   703→              v16.5: individual smart-block visibility is controlled inside
   704→              SmartBlocks via the homeLayout config (passed through). The
   705→              top-level isVisible('trending') etc. gates are handled by
   706→              SmartBlocks filtering its own blocks based on the same config. */}
   707→          <SmartBlocks onOpenProduct={onOpenProduct} limit={12} />
   708→
   709→          {/* Catalog CTA */}
   710→          <div className="px-4 md:px-6 py-3">
   711→            <div className="rounded-3xl gradient-soft p-6 md:p-8 flex flex-col sm:flex-row items-start sm:items-center gap-4">
   712→              <div className="flex-1">
   713→                <h3 className="text-xl font-bold mb-1">Полный каталог</h3>
   714→                <p className="text-sm text-muted-foreground">
   715→                  Сотни товаров с большими фото, ценами и быстрой покупкой в один клик.
   716→                </p>
   717→              </div>
   718→              <Button
   719→                onClick={() => onNavigate('catalog')}
   720→                className="rounded-full gradient-brand text-white font-semibold shadow-glow h-11 px-6"
   721→              >
   722→                Открыть каталог
   723→              </Button>
   724→            </div>
   725→          </div>
   726→        </>
   727→      )}
   728→
   729→      {/* v12.3: Feed preview block removed. The 999 CLUB module replaces
   730→          the old feed — the bottom nav now has a 💎 CLUB button. */}
   731→    </div>
   732→  )
