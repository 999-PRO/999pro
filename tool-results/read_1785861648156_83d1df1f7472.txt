   987→function ResponseCards({
   988→  cards,
   989→  onOpenProduct,
   990→}: {
   991→  cards: ChatCard[]
   992→  onOpenProduct?: (id: string) => void
   993→}) {
   994→  // Группируем product/similar_products карточки по category
   995→  // Если у карточки есть data.category — берём его; иначе — общий "Товары"
   996→  type Group = { title: string; emoji: string; products: any[] }
   997→  const productCards = cards.filter(c => c.kind === 'product' || c.kind === 'similar_products')
   998→  const otherCards = cards.filter(c => c.kind !== 'product' && c.kind !== 'similar_products')
   999→
  1000→  const groups: Group[] = []
  1001→  for (const card of productCards) {
  1002→    const items = Array.isArray(card.data) ? card.data : [card.data]
  1003→    for (const item of items) {
  1004→      const cat = item?.category || item?.categoryName || (card.kind === 'similar_products' ? 'Похожие товары' : 'Товары')
  1005→      let g = groups.find(x => x.title === cat)
  1006→      if (!g) {
  1007→        const emoji = categoryEmoji(cat)
  1008→        g = { title: cat, emoji, products: [] }
  1009→        groups.push(g)
  1010→      }
  1011→      g.products.push(item)
  1012→    }
  1013→  }
  1014→
  1015→  return (
  1016→    <div className="flex flex-col gap-3 pl-1">
  1017→      {groups.map((g, gi) => (
  1018→        <div key={gi} className="flex flex-col gap-2">

  1028→              <CompactProductCard
  1029→                key={i}
  1030→                product={p}
  1031→                onClick={onOpenProduct ? () => onOpenProduct(p.id || p.productId) : undefined}
  1032→              />
  1033→            ))}
  1034→          </div>
  1035→        </div>
  1036→      ))}

  1039→        if (card.kind === 'order_wizard') return <OrderWizard key={i} data={card.data} onSubmitted={() => {}} />

  1042→    </div>
  1043→  )

  1050→  if (c.includes('печат') || c.includes('print')) return '🖨'
  1051→  if (c.includes('визитк')) return '💳'
  1052→  if (c.includes('футбол') || c.includes('одежд')) return '👕'
  1053→  if (c.includes('кружк')) return '☕'
  1054→  if (c.includes('плакат') || c.includes('постер')) return '🖼'
  1055→  if (c.includes('буклет') || c.includes('листовк')) return '📄'
  1056→  if (c.includes('наклейк') || c.includes('стикер')) return '✨'
  1057→  if (c.includes('книг') || c.includes('журнал')) return '📚'
  1058→  if (c.includes('упаков')) return '📦'
  1059→  if (c.includes('сувенир')) return '🎀'
  1060→  return '🛍'

  1073→  const currencySymbol = (() => {
  1074→    const c = (product.currency || 'RUB').toString().toUpperCase()
  1075→    if (c === 'RUB' || c === 'РУБ') return '₽'
  1076→    if (c === 'USD') return '$'
  1077→    if (c === 'EUR') return '€'
  1078→    if (c === 'KZT') return '₸'
  1079→    if (c === 'UAH') return '₴'
  1080→    if (c === 'BYN') return 'Br'
  1081→    return c
  1082→  })()
  1083→  const isAction = product.isAction || product.discount

  1085→  const isPopular = product.isPopular

  1100→            src={typeof image === 'string' ? image : image.url}
  1101→            alt={name}
  1102→            className="absolute inset-0 w-full h-full object-cover group-hover:scale-105 transition-transform"
  1103→            loading="lazy"
