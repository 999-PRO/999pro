   131→   131→function useVoiceInput(opts: {
   132→   132→  onResult: (text: string) => void
   133→   133→  onInterim?: (text: string) => void
   134→   134→  onStart?: () => void
   135→   135→  onEnd?: () => void
   136→   136→  lang?: string
   137→   137→  continuous?: boolean
   138→   138→}) {
   139→   139→  const [listening, setListening] = useState(false)
   140→   140→  const recRef = useRef<any>(null)
   141→   141→  const startingRef = useRef(false)
   142→   142→  // v5: фиксируем onEnd, чтобы вызывать его когда recognition завершился сам
   143→   143→  // (например, по таймауту браузера). Это нужно для логики «Вы закончили?».
   144→   144→  const onEndRef = useRef<() => void>(() => {})
   145→   145→  onEndRef.current = opts.onEnd || (() => {})
   146→   146→
   147→   147→  const stop = useCallback(() => {
   148→   148→    startingRef.current = false
   149→   149→    if (recRef.current) {
   150→   150→      try {
   151→   151→        recRef.current.onresult = null
   152→   152→        recRef.current.onerror = null
   153→   153→        recRef.current.onend = null
   154→   154→        recRef.current.onstart = null
   155→   155→        recRef.current.abort?.()
   156→   156→        recRef.current.stop?.()
   157→   157→      } catch {}
   158→   158→      recRef.current = null
   159→   159→    }
   160→   160→    setListening(false)
   161→   161→  }, [])
   162→   162→
   163→   163→  const start = useCallback(() => {
   164→   164→    if (typeof window === 'undefined') return
   165→   165→    if (listening || startingRef.current) return
   166→   166→    if (recRef.current) return
   167→   167→    const SR = (window as any).SpeechRecognition || (window as any).webkitSpeechRecognition
   168→   168→    if (!SR) {
   169→   169→      alert('Голосовой ввод не поддерживается. Используйте Chrome, Edge или Safari.')
   170→   170→      return
   171→   171→    }
   172→   172→    startingRef.current = true
   173→   173→    const rec = new SR()
   174→   174→    rec.lang = opts.lang || 'ru-RU'
   175→   175→    // v5: continuous = true — чтобы recognition не закрывался после каждой фразы.
   176→   176→    // Это позволяет нам самим управлять моментом «пользователь замолчал».
   177→   177→    rec.continuous = opts.continuous !== false
   178→   178→    rec.interimResults = true
   179→   179→    rec.maxAlternatives = 1
   180→   180→    rec.onstart = () => {
   181→   181→      startingRef.current = false
   182→   182→      setListening(true)
   183→   183→      opts.onStart?.()
   184→   184→    }
   185→   185→    rec.onerror = (e: any) => {
   186→   186→      startingRef.current = false
   187→   187→      setListening(false)
   188→   188→      console.warn('[VoiceInput] error:', e?.error)
   189→   189→    }
   190→   190→    rec.onend = () => {
   191→   191→      startingRef.current = false
   192→   192→      setListening(false)
   193→   193→      // v24.3 BUGFIX: clear recRef.current so subsequent start() calls can
   194→   194→      // instantiate a fresh SpeechRecognition. Previously, on desktop Chrome,
   195→   195→      // recognition.onend fired after each utterance/silence but recRef.current
   196→   196→      // was never cleared — so the guard `if (recRef.current) return` in
   197→   197→      // start() blocked the next start() call, and follow-up questions were
   198→   198→      // never captured. Mobile Chrome with continuous=true keeps recognition
   199→   199→      // alive across utterances (onend doesn't fire mid-conversation), which
   200→   200→      // is why mobile worked but desktop didn't.
   201→   201→      recRef.current = null
   202→   202→      // v5: браузер сам завершил recognition (timeout или no-speech).
   203→   203→      // Уведомляем caller, чтобы он мог решить, перезапускать или нет.
   204→   204→      onEndRef.current?.()
   205→   205→    }
   206→   206→    rec.onresult = (e: any) => {
   207→   207→      let finalText = ''
   208→   208→      let interimText = ''
   209→   209→      for (let i = e.resultIndex; i < e.results.length; i++) {
   210→   210→        const r = e.results[i]
   211→   211→        if (r.isFinal) finalText += r[0].transcript
   212→   212→        else interimText += r[0].transcript
   213→   213→      }
   214→   214→      if (interimText) opts.onInterim?.(interimText)
   215→   215→      if (finalText) {
   216→   216→        const cleaned = finalText.trim()
   217→   217→        if (cleaned) opts.onResult(cleaned)
   218→   218→      }
   219→   219→    }
   220→   220→    recRef.current = rec
   221→   221→    try {
   222→   222→      rec.start()
   223→   223→    } catch (err) {
   224→   224→      startingRef.current = false
   225→   225→      console.warn('[VoiceInput] start() threw:', err)
   226→   226→    }
   227→   227→  }, [opts, listening, stop])
   228→   228→
   229→   229→  useEffect(() => () => stop(), [stop])
   230→   230→  return { listening, start, stop }
   231→   231→}
