    23→  '/conversations',
    24→  requireAuth,
    25→  asyncHandler(async (req: AuthedRequest, res) => {
    26→    const { participantId } = startConversationSchema.parse(req.body)
    27→    const meId = req.user!.id
    28→    if (participantId === meId) {
    29→      return res.status(400).json({ error: 'Cannot start conversation with yourself' })
    30→    }
    31→
    32→    // Phase 8.1: cannot start conversation with a soft-deleted user.
    33→    const participant = await prisma.user.findUnique({ where: { id: participantId } })
    34→    if (!participant || participant.deletedAt) {
    35→      return res.status(404).json({ error: 'User not found' })
    36→    }
    37→
    38→    const existingConversations = await prisma.conversation.findMany({
    39→      where: { participants: { some: { userId: meId } } },
    40→      include: { participants: { select: { userId: true } } },
    41→      take: 200,
    42→    })
    43→
    44→    const existing = existingConversations.find(
    45→      (c) =>
    46→        c.participants.length === 2 &&
    47→        c.participants.some((p) => p.userId === meId) &&
    48→        c.participants.some((p) => p.userId === participantId),
    49→    )
    50→
    51→    if (existing) {
    52→      const fullConv = await prisma.conversation.findUnique({
    53→        where: { id: existing.id },
    54→        include: {
    55→          participants: { include: { user: { select: userSelect } } },
    56→          messages: { orderBy: { createdAt: 'desc' }, take: 1 },
    57→        },
    58→      })
    59→      return res.json({ conversation: formatConversation(fullConv, meId) })
    60→    }
    61→
    62→    const conv = await prisma.conversation.create({
    63→      data: {
    64→        participants: { create: [{ userId: meId }, { userId: participantId }] },
    65→      },
    66→      include: {
    67→        participants: { include: { user: { select: userSelect } } },
    68→        messages: { orderBy: { createdAt: 'desc' }, take: 1 },
    69→      },
    70→    })

    75→// ============================================================================
    76→//  GET /api/chat/users — list of users available for chatting.
    77→//
    78→//  PRIVACY POLICY (revised for marketplace use case):
    79→//    A marketplace chat requires users to be able to find any other
    80→//    registered user by username and start a conversation with them
    81→//    (buyer ↔ seller, support ↔ customer, user ↔ user). The previous
    82→//    policy (regular users see ONLY admins + existing conversation
    83→//    partners) broke this core flow — users could never find a
    84→//    newly-registered contact by nickname.
    85→//
    86→//    New policy:
    87→//      - ALL authenticated callers can see ALL registered users
    88→//        (excluding themselves) by PUBLIC fields only:
    89→//          username, displayName, avatar, isOnline, lastSeen, role
    90→//      - Email and phone are NEVER returned to non-admin callers
    91→//        (sensitive PII — only admins see them, and only when searching).
    92→//      - When a search query is provided, search is performed across
    93→//        username + displayName (public fields). Admins additionally
    94→//        can search by email and phone.
    95→//      - The admin is ALWAYS pinned to the top of the list with
    96→//        `isSupport: true` so the UI can render them with a distinct
    97→//        card / colour.
    98→//
    99→//  The list is ordered by:
   100→//    1. Admins (pinned, isSupport: true)
   101→//    2. Users with an existing 1-on-1 conversation with the caller
   102→//       (most recent message first)
   103→//    3. Online users
   104→//    4. Most recently active users (lastSeen desc)
   105→router.get(
   106→  '/users',
   107→  requireAuth,
   108→  asyncHandler(async (req: AuthedRequest, res) => {
   109→    const meId = req.user!.id
   110→    const isAdmin = req.user!.role === 'admin'
   111→    const limit = Math.min(Math.max(Number(req.query.limit ?? 100) || 100, 1), 500)
   112→    const offset = Math.max(Number(req.query.offset ?? 0) || 0, 0)
   113→    const search = (req.query.q as string | undefined)?.trim()
   114→
   115→    // ----------------------------------------------------------------
   116→    // Build WHERE clause.
   117→    // NEW BEHAVIOUR: users see ONLY:
   118→    //   - Admins (always visible — they're the support channel)
   119→    //   - Users they already have a conversation with
   120→    // When searching, also include ALL users matching the search query
   121→    // (so the user can discover and start a new conversation).
   122→    // ----------------------------------------------------------------
   123→    // Phase 8.1: exclude soft-deleted users from chat candidates.
   124→    const where: any = { id: { not: meId }, deletedAt: null }
   125→
   126→    // Fetch the caller's existing conversation partners FIRST.
   127→    const myConvPartners = await prisma.conversationParticipant.findMany({
   128→      where: { userId: meId },
   129→      select: {
   130→        conversationId: true,
   131→        conversation: {
   132→          select: {
   133→            type: true,
   134→            participants: { select: { userId: true } },
   135→            messages: { orderBy: { createdAt: 'desc' }, take: 1, select: { createdAt: true } },
   136→          },
   137→        },
   138→      },
   139→    })
   140→    // Build a map: partnerUserId -> { lastMessageAt, type }
   141→    const partnerInfo = new Map<string, { lastMessageAt: Date; type: string }>()
   142→    const partnerIds = new Set<string>()
   143→    for (const cp of myConvPartners) {
   144→      const conv = cp.conversation
   145→      const otherUserId = conv.participants.find((p) => p.userId !== meId)?.userId
   146→      if (!otherUserId) continue
   147→      const lastAt = conv.messages[0]?.createdAt || new Date(0)
   148→      partnerInfo.set(otherUserId, { lastMessageAt: lastAt as Date, type: conv.type })
   149→      partnerIds.add(otherUserId)
   150→    }
   151→
   152→    if (search && search.length >= 1) {
   153→      // SEARCH MODE: search across ALL users (username + displayName).
   154→      // This allows discovering new users to start a conversation.
   155→      const searchOR: any[] = [
   156→        { username: { contains: search } },
   157→        { displayName: { contains: search } },
   158→      ]
   159→      if (isAdmin) {
   160→        searchOR.push({ email: { contains: search } })
   161→        if (search.startsWith('+') || /^\d/.test(search)) {
   162→          searchOR.push({ phone: { contains: search } })
   163→        }
   164→      }
   165→      where.OR = searchOR
   166→    } else {
   167→      // v11-fix: NO SEARCH — show ONLY existing conversation partners.
   168→      // Previously this also included `{ role: 'admin' }` which caused new
   169→      // users to see ALL admins in the system (even ones they never talked
   170→      // to). Now we show only users the current user has an actual
   171→      // conversation with. The admin support chat (auto-created at
   172→      // registration) will be in partnerIds, so the user still sees their
   173→      // admin — but only THAT admin, not every admin in the system.
   174→      //
   175→      // Edge case: if the user has no conversations at all (brand new
   176→      // account, registration just completed), partnerIds is empty and the
   177→      // result is empty. The frontend ChatView handles this by showing the
   178→      // "start new chat" search, which uses the search branch above.
   179→      if (partnerIds.size === 0) {
   180→        return res.json({ users: [], total: 0, limit, offset })
   181→      }
   182→      where.id = { in: [...partnerIds] }
   183→    }
   184→
   185→    // Fetch candidate users
   186→    const candidates = await prisma.user.findMany({
   187→      where,
   188→      select: {
   189→        id: true, username: true, displayName: true, avatar: true,
   190→        isOnline: true, lastSeen: true, role: true,
   191→        ...(isAdmin && search ? { email: true, phone: true } : {}),
   192→      },
   193→      take: search ? 200 : limit + offset,
   194→      orderBy: [{ isOnline: 'desc' }, { lastSeen: 'desc' }],
   195→    })
   196→
   197→    // If searching with Cyrillic or mixed case, do a case-insensitive filter
   198→    let filtered = candidates
   199→    if (search) {
   200→      const needle = search.toLowerCase()
   201→      filtered = candidates.filter((u) => {
   202→        const username = (u.username || '').toLowerCase()
   203→        const displayName = (u.displayName || '').toLowerCase()
   204→        if (username.includes(needle) || displayName.includes(needle)) return true
   205→        if (isAdmin) {
   206→          const email = (u.email || '').toLowerCase()
   207→          const phone = (u.phone || '').toLowerCase()
   208→          if (email.includes(needle) || phone.includes(needle)) return true
   209→        }
   210→        return false
   211→      })
   212→    }
   213→
   214→    // partnerInfo already fetched above — reuse it.
   215→
   216→    // Build the final list — publicUser shape + hasConversation flag.
   217→    // v9-audit-fix: privacy — for users the caller has NO conversation with
   218→    // (strangers discovered via search), strip isOnline/lastSeen to prevent
   219→    // activity surveillance. Admins and existing partners keep full fields.
   220→    const shaped = filtered.map((u) => {
   221→      const info = partnerInfo.get(u.id)
   222→      const isPartner = !!info
   223→      const isAdminUser = u.role === 'admin'
   224→      const isStranger = !isPartner && !isAdminUser
   225→      return {
   226→        id: u.id,
   227→        username: u.username,
   228→        displayName: u.displayName,
   229→        avatar: u.avatar,
   230→        // v9-audit-fix: hide online status + last seen from strangers
   231→        isOnline: isStranger ? null : u.isOnline,
   232→        lastSeen: isStranger ? null : u.lastSeen,
   233→        role: u.role,
   234→        isSupport: isAdminUser, // UI hint: render with admin styling
   235→        hasConversation: isPartner,
   236→        conversationType: info?.type || null,
   237→        lastMessageAt: info?.lastMessageAt || null,
   238→      }
   239→    })
   240→
   241→    // Sort: admins first (pinned support), then users with existing
   242→    // conversations (by lastMessageAt desc), then online, then by lastSeen.
   243→    shaped.sort((a, b) => {
   244→      // Admins always first
   245→      if (a.isSupport && !b.isSupport) return -1
   246→      if (!a.isSupport && b.isSupport) return 1
   247→      // Both have conversations → most recent message first
   248→      if (a.hasConversation && b.hasConversation) {
   249→        const aTime = a.lastMessageAt?.getTime() || 0
   250→        const bTime = b.lastMessageAt?.getTime() || 0
   251→        return bTime - aTime
   252→      }
   253→      // One has a conversation → that one comes first
   254→      if (a.hasConversation && !b.hasConversation) return -1
   255→      if (!a.hasConversation && b.hasConversation) return 1
   256→      // Neither has a conversation → online first, then by lastSeen
   257→      if (a.isOnline !== b.isOnline) return a.isOnline ? -1 : 1
   258→      const aSeen = a.lastSeen?.getTime() || 0
   259→      const bSeen = b.lastSeen?.getTime() || 0
   260→      return bSeen - aSeen
   261→    })
   262→
   263→    const paged = shaped.slice(offset, offset + limit)
   264→    res.json({ users: paged, total: shaped.length, limit, offset })
   265→  }),
   266→)

   268→// GET /api/chat/conversations
   269→router.get(
   270→  '/conversations',
   271→  requireAuth,
   272→  asyncHandler(async (req: AuthedRequest, res) => {
   273→    const meId = req.user!.id
   274→    const isAdmin = req.user!.role === 'admin'
   275→    const limit = Math.min(Math.max(Number(req.query.limit ?? 50) || 50, 1), 200)
   276→    const offset = Math.max(Number(req.query.offset ?? 0) || 0, 0)
   277→
   278→    // v11-fix: for admins, hide "empty" support conversations — ones where
   279→    // the only message is the welcome message sent by the admin themselves
   280→    // (i.e. the user never replied). This keeps the admin's chat list clean:
   281→    // they only see users who have actually engaged in conversation.
   282→    // Regular users always see all their conversations (including the
   283→    // welcome-only support chat with the admin).
   284→    //
   285→    // v13.0 (audit P0-2 fix): the previous version built a raw SQL string
   286→    // via string interpolation with single-quote escaping. While cuid()
   287→    // values are safe today, the pattern was a textbook SQL-injection
   288→    // waiting to happen. Now uses Prisma's parameterised SQL fragment.
   289→    const adminFilter = isAdmin
   290→      ? Prisma.sql`AND NOT (
   291→           SELECT COUNT(*) FROM Message m
   292→           WHERE m.conversationId = c.id
   293→             AND m.senderId <> ${meId}
   294→         ) = 0
   295→           OR c.type <> 'support'`
   296→      : Prisma.empty
   297→
   298→    // Build the query with the admin filter applied
   299→    const conversations = isAdmin
   300→      ? await prisma.$queryRaw`
   301→          SELECT c.* FROM Conversation c
   302→          WHERE c.id IN (
   303→            SELECT conversationId FROM ConversationParticipant WHERE userId = ${meId}
   304→          )
   305→          ${adminFilter}
   306→          ORDER BY c.type DESC, c.updatedAt DESC
   307→          LIMIT ${limit} OFFSET ${offset}
   308→        `
   309→      : await prisma.conversation.findMany({
   310→          where: { participants: { some: { userId: meId } } },
   311→          include: {
   312→            participants: { include: { user: { select: userSelect } } },
   313→            messages: { orderBy: { createdAt: 'desc' }, take: 1 },
   314→          },
   315→          orderBy: [
   316→            { type: 'desc' },
   317→            { updatedAt: 'desc' },
   318→          ],
   319→          take: limit,
   320→          skip: offset,
   321→        })
   322→
   323→    // For the admin raw query, we need to load participants + last message separately
   324→    let formattedConversations: any[]
   325→    let total: number
   326→    if (isAdmin) {
   327→      const rawConvIds = (conversations as Array<{ id: string }>).map((c) => c.id)
   328→      if (rawConvIds.length === 0) {
   329→        formattedConversations = []
   330→        total = 0
   331→      } else {
   332→        const fullConvs = await prisma.conversation.findMany({
   333→          where: { id: { in: rawConvIds } },
   334→          include: {
   335→            participants: { include: { user: { select: userSelect } } },
   336→            messages: { orderBy: { createdAt: 'desc' }, take: 1 },
   337→          },
   338→          orderBy: [
   339→            { type: 'desc' },
   340→            { updatedAt: 'desc' },
   341→          ],
   342→        })
   343→        // Preserve the order from the raw query (which already sorted by type DESC, updatedAt DESC)
   344→        const orderMap = new Map(rawConvIds.map((id, i) => [id, i]))
   345→        fullConvs.sort((a, b) => (orderMap.get(a.id) ?? 0) - (orderMap.get(b.id) ?? 0))
   346→        formattedConversations = fullConvs.map((c) => formatConversation(c, meId))
   347→        // Count total (with the same admin filter applied)
   348→        const totalResult = await prisma.$queryRaw`
   349→          SELECT COUNT(*) as cnt FROM Conversation c
   350→          WHERE c.id IN (
   351→            SELECT conversationId FROM ConversationParticipant WHERE userId = ${meId}
   352→          )
   353→          ${adminFilter}
   354→        `
   355→        total = Number((totalResult as Array<{ cnt: bigint }>)[0]?.cnt ?? 0)
   356→      }
   357→    } else {
   358→      formattedConversations = (conversations as any[]).map((c) => formatConversation(c, meId))
   359→      total = await prisma.conversation.count({
   360→        where: { participants: { some: { userId: meId } } },
   361→      })
   362→    }
   363→
   364→    res.json({
   365→      items: formattedConversations,
   366→      total,
   367→      limit,
   368→      offset,
   369→    })
   370→  }),
   371→)

   373→// GET /api/chat/unread-counts
   374→// C-HIGH-005: aggregate unread message counts per-conversation (and total)
   375→// for the current user. Used by clients to sync unread badges across
   376→// multiple devices — when the user reads a conversation on desktop, mobile
   377→// can poll this endpoint (or receive a socket event) and clear its badge.
   378→//
   379→// "Unread" here is intentionally simplified: a message is counted as unread
   380→// if `isRead=false` AND `senderId != meId`. The `deletedFor` JSON array
   381→// (per-user soft-delete) is NOT filtered here because SQLite can't index
   382→// JSON arrays efficiently — if we filtered client-side we'd return an
   383→// inflated count, so for now we accept a minor over-count for messages the
   384→// user has deleted-for-me. A follow-up should migrate `deletedFor` to a
   385→// join table for O(1) filtering.
   386→//
   387→// NOTE: route declared BEFORE `/conversations/:id/*` so Express matches
   388→// `/unread-counts` literally and never tries to parse it as a conversation id.
   389→router.get(
   390→  '/unread-counts',
   391→  requireAuth,
   392→  asyncHandler(async (req: AuthedRequest, res) => {
   393→    const meId = req.user!.id
   394→    // v10-fix: filter by conversation membership — previously this query
   395→    // returned unread counts for ALL conversations in the DB (privacy leak).
   396→    // Now we only count messages in conversations the user is a participant of.
   397→    const myConvIds = await prisma.conversationParticipant.findMany({
   398→      where: { userId: meId },
   399→      select: { conversationId: true },
   400→    })
   401→    const convIdList = myConvIds.map((cp) => cp.conversationId)
   402→    if (convIdList.length === 0) {
   403→      return res.json({ byConversation: {}, total: 0 })
   404→    }
   405→    const rows = await prisma.message.groupBy({
   406→      by: ['conversationId'],
   407→      where: {
   408→        isRead: false,
   409→        senderId: { not: meId },
   410→        conversationId: { in: convIdList },
   411→      },
   412→      _count: { _all: true },
   413→    })
   414→    const byConversation: Record<string, number> = {}
   415→    let total = 0
   416→    for (const r of rows) {
   417→      byConversation[r.conversationId] = r._count._all
   418→      total += r._count._all
   419→    }
   420→    res.json({ byConversation, total })
   421→  }),

   424→// GET /api/chat/conversations/:id/messages
   425→router.get(
   426→  '/conversations/:id/messages',
   427→  requireAuth,
   428→  asyncHandler(async (req: AuthedRequest, res) => {
   429→    const meId = req.user!.id
   430→    const conv = await prisma.conversation.findUnique({
   431→      where: { id: req.params.id },
   432→      include: { participants: { select: { userId: true } } },
   433→    })
   434→    if (!conv) return res.status(404).json({ error: 'Conversation not found' })
   435→    const isMember = conv.participants.some((p) => p.userId === meId)
   436→    if (!isMember) return res.status(403).json({ error: 'Forbidden' })
   437→
   438→    const limit = Math.min(Math.max(Number(req.query.limit ?? 100) || 100, 1), 500)
   439→    const offset = Math.max(Number(req.query.offset ?? 0) || 0, 0)
   440→
   441→    // C-HIGH-003 fix: cursor-based pagination (backward compatible with offset).
   442→    // ?before=<messageId> — returns messages older than the given message.
   443→    // ?limit=100 — page size (default 100, max 500).
   444→    // When `before` is supplied, we use cursor pagination (keyset) which is
   445→    // O(log N) and stable under concurrent inserts, unlike offset which is
   446→    // O(offset+limit) and shifts when new messages arrive mid-pagination.
   447→    // When `before` is absent, the original offset path is preserved so
   448→    // existing clients keep working unchanged.
   449→    const before = req.query.before as string | undefined
   450→    if (before) {
   451→      const cursorMsg = await prisma.message.findUnique({
   452→        where: { id: before },
   453→        select: { createdAt: true },
   454→      })
   455→      const cursorCreatedAt = cursorMsg?.createdAt
   456→      const rawMessages = await prisma.message.findMany({
   457→        where: {
   458→          conversationId: conv.id,
   459→          ...(cursorCreatedAt ? { createdAt: { lt: cursorCreatedAt } } : {}),
   460→        },
   461→        orderBy: { createdAt: 'desc' }, // newest first
   462→        take: limit,
   463→        include: {
   464→          sender: { select: userSelect },
   465→          replyTo: { include: { sender: { select: userSelect } } },
   466→          forwardedFrom: { include: { sender: { select: userSelect } } },
   467→        },
   468→      })
   469→      // Reverse to ascending order for client display
   470→      rawMessages.reverse()
   471→      // Filter out messages deleted for the viewer
   472→      const visible = rawMessages.filter((m) => {
   473→        const deletedFor: string[] = (() => {
   474→          try { return JSON.parse(m.deletedFor || '[]') } catch { return [] }
   475→        })()
   476→        return !deletedFor.includes(meId)
   477→      })
   478→      res.json({
   479→        items: visible.map((m) => serialiseMessage(m, meId)),
   480→        total: visible.length,
   481→        limit,
   482→        offset: 0,
   483→        hasMore: visible.length === limit,
   484→      })
   485→      return
   486→    }
   487→
   488→    // Fetch oldest-first (skip messages deleted for me)
   489→    const messages = await prisma.message.findMany({
   490→      where: { conversationId: conv.id },
   491→      orderBy: { createdAt: 'asc' },
   492→      take: limit,
   493→      skip: offset,
   494→      include: {
   495→        sender: { select: userSelect },
   496→        replyTo: { include: { sender: { select: userSelect } } },
   497→        forwardedFrom: { include: { sender: { select: userSelect } } },
   498→      },
   499→    })
   500→    // Filter out messages deleted for the viewer (they shouldn't see them at all)
   501→    const visible = messages.filter((m) => {
   502→      const deletedFor: string[] = (() => {
   503→        try { return JSON.parse(m.deletedFor || '[]') } catch { return [] }
   504→      })()
   505→      return !deletedFor.includes(meId)
   506→    })
   507→    res.json({
   508→      items: visible.map((m) => serialiseMessage(m, meId)),
   509→      total: visible.length,
   510→      limit,
   511→      offset,
   512→    })
   513→  }),
   514→)

   526→  asyncHandler(async (req: AuthedRequest, res) => {
   527→    const meId = req.user!.id
   528→    const conv = await prisma.conversation.findUnique({
   529→      where: { id: req.params.id },
   530→      include: { participants: { select: { id: true, userId: true } } },
   531→    })
   532→    if (!conv) return res.status(404).json({ error: 'Conversation not found' })
   533→
   534→    const isMember = conv.participants.some((p) => p.userId === meId)
   535→    if (!isMember) return res.status(403).json({ error: 'Forbidden' })
   536→

   538→    // channel and must always be available to the user.
   539→    if (conv.type === 'support') {

   541→    }

   543→    // SECURITY / DATA-LOSS FIX (was):
   544→    // The previous implementation called `prisma.message.deleteMany({ where: { conversationId: conv.id } })`
   545→    // — this DELETED EVERY MESSAGE IN THE CONVERSATION FOR ALL PARTICIPANTS,
   546→    // including the other party's received messages. User A could unilaterally
   547→    // destroy user B's chat history without consent or audit trail.
   548→    //
   549→    // New behaviour: "delete" = soft-delete for the caller only.
   550→    //   1. Mark every message in the conversation as `deletedFor: [meId]`
   551→    //      (the existing soft-delete mechanism used by `message:delete` socket event).
   552→    //      The caller no longer sees these messages, but the other participant
   553→    //      still has their full history.
   554→    //   2. Remove the caller from `ConversationParticipant` so the conversation
   555→    //      disappears from their chat list.
   556→    //   3. If no participants remain (both users deleted), then we hard-delete
   557→    //      the conversation AND its messages (no orphan data).
   558→    //
   559→    // This mirrors how WhatsApp / Telegram handle "delete conversation":
   560→    // the action only affects the caller's view, not the other party's.
   561→    //
   562→    // v9-audit-fix: replaced N+1 per-message UPDATE loop with a single bulk
   563→    // SQL UPDATE using SQLite json_insert. Wrapped the whole operation in a
   564→    // $transaction so a crash between steps can't leave the conversation
   565→    // visible but with all messages hidden.
   566→    await prisma.$transaction(async (tx) => {
   567→      // Bulk-update: add meId to every message's deletedFor JSON array (if not
   568→      // already present) in a single SQL statement.
   569→      await tx.$executeRaw`
   570→        UPDATE "Message"
   571→        SET "deletedFor" = CASE
   572→          WHEN "deletedFor" IS NULL OR "deletedFor" = '' OR "deletedFor" = '[]' THEN json_array(${meId})
   573→          WHEN EXISTS (SELECT 1 FROM json_each("deletedFor") WHERE value = ${meId}) THEN "deletedFor"
   574→          ELSE json_insert("deletedFor", '$[#]', ${meId})
   575→        END
   576→        WHERE "conversationId" = ${conv.id}
   577→      `
   578→
   579→      // Remove the caller from the conversation's participant list.
   580→      await tx.conversationParticipant.deleteMany({
   581→        where: { conversationId: conv.id, userId: meId },
   582→      })
   583→
   584→      // If no participants remain, hard-delete the conversation + messages
   585→      // (orphan cleanup — no one will ever see this data again).
   586→      const remaining = await tx.conversationParticipant.count({
   587→        where: { conversationId: conv.id },
   588→      })
   589→      if (remaining === 0) {
   590→        await tx.message.deleteMany({ where: { conversationId: conv.id } })
   591→        await tx.conversation.delete({ where: { id: conv.id } })
   592→      }
   593→    })
   594→
   595→    res.json({ ok: true })
   596→  }),

   972→function formatConversation(c: any, meId: string) {
   973→  const others = c.participants
   974→    .filter((p: any) => p.userId !== meId)
   975→    .map((p: any) => p.user)
   976→  const other = others[0]
   977→  const lastMessage = c.messages?.[0]
   978→  return {
   979→    id: c.id,
   980→    // 'direct' | 'support' — support conversations are pinned to the top
   981→    // of the chat list and cannot be deleted by the user.
   982→    type: c.type || 'direct',
   983→    participant: other
   984→      ? {
   985→          id: other.id,
   986→          username: other.username,
   987→          displayName: other.displayName,
   988→          avatar: other.avatar,
   989→          isOnline: other.isOnline,
   990→          lastSeen: other.lastSeen,
   991→        }
   992→      : null,
   993→    participants: others.map((u: any) => ({
   994→      id: u.id,
   995→      username: u.username,
   996→      displayName: u.displayName,
   997→      avatar: u.avatar,
   998→      isOnline: u.isOnline,
   999→      lastSeen: u.lastSeen,
  1000→    })),
  1001→    lastMessage: lastMessage
  1002→      ? {
  1003→          id: lastMessage.id,
  1004→          content: lastMessage.deletedForAll ? null : lastMessage.content,
  1005→          mediaUrl: lastMessage.deletedForAll ? null : lastMessage.mediaUrl,
  1006→          mediaType: lastMessage.deletedForAll ? null : lastMessage.mediaType,
  1007→          createdAt: lastMessage.createdAt,
  1008→          senderId: lastMessage.senderId,
  1009→          deletedForAll: lastMessage.deletedForAll,
  1010→        }
  1011→      : null,
  1012→    updatedAt: c.updatedAt,
  1013→    createdAt: c.createdAt,
  1014→  }

  1046→    // Find an admin to assign as the support agent. Pick the one with the
  1047→    // fewest existing conversations (load balancing).
  1048→    // Phase 8.1: only assign active (non-deleted) admins.
  1049→    const admins = await prisma.user.findMany({
  1050→      where: { role: 'admin', deletedAt: null },
  1051→      select: {
  1052→        id: true,
  1053→        username: true,
  1054→        displayName: true,
  1055→        avatar: true,
  1056→        isOnline: true,
  1057→        _count: { select: { conversations: true } },
  1058→      },
  1059→    })
  1060→
  1061→    if (admins.length === 0) {
  1062→      // No admin exists yet — return a friendly error. The first-run
  1063→      // admin wizard in Studio will create one.
  1064→      return res.status(503).json({
  1065→        error: 'Команда поддержки сейчас недоступна. Попробуйте позже или напишите на support@999.pro',
  1066→      })
  1067→    }
  1068→
  1069→    // Pick the admin with the fewest conversations (load balancing).
  1070→    // Sort by _count.conversations ascending, then by isOnline (online first).
  1071→    admins.sort((a, b) => {
  1072→      const aCount = a._count.conversations
  1073→      const bCount = b._count.conversations
  1074→      if (aCount !== bCount) return aCount - bCount
  1075→      // Prefer online admins
  1076→      return (b.isOnline ? 1 : 0) - (a.isOnline ? 1 : 0)
  1077→    })
  1078→    const supportAdmin = admins[0]
  1079→
  1080→    // Check if a conversation already exists between this user and the admin.
  1081→    const existingConversations = await prisma.conversation.findMany({
  1082→      where: { participants: { some: { userId: meId } } },
  1083→      include: { participants: { select: { userId: true } } },
  1084→      take: 200,
  1085→    })
  1086→
  1087→    const existing = existingConversations.find(
  1088→      (c) =>
  1089→        c.participants.length === 2 &&
  1090→        c.participants.some((p) => p.userId === meId) &&
  1091→        c.participants.some((p) => p.userId === supportAdmin.id),
  1092→    )
  1093→
  1094→    if (existing) {
  1095→      const fullConv = await prisma.conversation.findUnique({
  1096→        where: { id: existing.id },
  1097→        include: {
  1098→          participants: { include: { user: { select: userSelect } } },
  1099→          messages: { orderBy: { createdAt: 'desc' }, take: 1 },
  1100→        },
  1101→      })
  1102→      return res.json({ conversation: formatConversation(fullConv, meId) })
  1103→    }
  1104→
  1105→    // Create a new support conversation. The `type: 'support'` flag makes
  1106→    // this conversation pinned to the top of the user's chat list and
  1107→    // prevents the user from deleting it.
  1108→    const conv = await prisma.conversation.create({
  1109→      data: {
  1110→        type: 'support',
  1111→        participants: { create: [{ userId: meId }, { userId: supportAdmin.id }] },
  1112→      },
  1113→      include: {
  1114→        participants: { include: { user: { select: userSelect } } },
  1115→        messages: { orderBy: { createdAt: 'desc' }, take: 1 },
  1116→      },
  1117→    })
  1118→
  1119→    // v25.3 (TZ task #3): automatic welcome message removed.
  1120→    //
  1121→    // Previously the backend created a "Здравствуйте! Я из команды поддержки…"
  1122→    // message on every new support conversation, which counted as a test /
  1123→    // demo message polluting the chat list. Per the technical spec, the chat
  1124→    // must open empty after launch — no demo / sample / placeholder messages
  1125→    // of any kind.
  1126→    //
  1127→    // The empty support conversation is returned as-is; the support view's
  1128→    // placeholder input ("Напишите сообщение…") makes the next step obvious
  1129→    // to the user.
  1130→
  1131→    res.status(201).json({ conversation: formatConversation(conv, meId) })
  1132→  }),
  1133→)
  1134→
