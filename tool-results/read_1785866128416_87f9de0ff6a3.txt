    83→/**
    84→ * CORS origin checker.
    85→ *
    86→ * v25.2-CORS-FIX: When origin is NOT allowed, we now pass an Error to the
    87→ * callback (callback(err)) instead of callback(null, false). This makes
    88→ * cors() respond with 500 + error message instead of passing the request
    89→ * through to Express (which would respond 200 + Allow header without any
    90→ * Access-Control-Allow-Origin, confusing the browser).
    91→ *
    92→ * In practice this shouldn't matter anymore because the frontend now uses
    93→ * relative /api/* paths (no cross-origin), but this is defense-in-depth:
    94→ * if someone manually calls the backend from a different origin, they get
    95→ * a clear error instead of a silent 200-without-CORS-header.
    96→ *
    97→ * In development: allow any origin on localhost or LAN IPs (so the app works
    98→ * when opened from a phone at http://192.168.x.x:3000), plus any *.space-z.ai
    99→ * subdomain (the sandbox/preview gateway) and the wildcard `*` if configured.
   100→ *
   101→ * In production: only allow the configured CLIENT_ORIGIN allowlist
   102→ * (or everything if CLIENT_ORIGIN contains `*`).
   103→ */
   104→function corsOrigin(origin: string | undefined, callback: (err: Error | null, ok?: boolean) => void) {
   105→  // Allow same-origin requests (no Origin header, e.g. curl, server-to-server)
   106→  if (!origin) return callback(null, true)
   107→
   108→  // Wildcard shortcut: if CLIENT_ORIGIN contains `*`, allow every origin.
   109→  // In production this is REJECTED — wildcards are dev-only.
   110→  if (CLIENT_ORIGIN.includes('*')) {
   111→    if (isProd) {
   112→      logger.warn('CORS CLIENT_ORIGIN=* forbidden in production', { module: 'cors', origin })
   113→      return callback(new Error(`CORS: wildcard origin forbidden in production (got ${origin})`))
   114→    }
   115→    return callback(null, true)
   116→  }
   117→
   118→  // Production: strict allowlist
   119→  if (isProd) {
   120→    if (CLIENT_ORIGIN.includes(origin)) return callback(null, true)
   121→    // Also allow any *.space-z.ai preview subdomain in prod (sandbox gateway)
   122→    try {
   123→      const url = new URL(origin)
   124→      if (url.hostname === 'space-z.ai' || url.hostname.endsWith('.space-z.ai')) {
   125→        return callback(null, true)
   126→      }
   127→    } catch {}
   128→    logger.warn('CORS origin rejected (production allowlist)', {
   129→      module: 'cors',
   130→      origin,
   131→      allowlist: CLIENT_ORIGIN,
   132→    })
   133→    return callback(new Error(`CORS: origin ${origin} not in CLIENT_ORIGIN allowlist`))
   134→  }
   135→
   136→  // Development: allow localhost + LAN IPs on any port
   137→  try {
   138→    const url = new URL(origin)
   139→    const host = url.hostname
   140→    if (
   141→      host === 'localhost' ||
   142→      host === '127.0.0.1' ||
   143→      /^(192\.168\.|10\.|172\.(1[6-9]|2[0-9]|3[01])\.|169\.254\.)/.test(host) ||
   144→      host.endsWith('.local')
   145→    ) {
   146→      return callback(null, true)
   147→    }
   148→    // Sandbox preview gateway: any *.space-z.ai subdomain (e.g. preview-<bot-id>.space-z.ai)
   149→    if (host === 'space-z.ai' || host.endsWith('.space-z.ai')) {
   150→      return callback(null, true)
   151→    }
   152→  } catch {}
   153→  // Fallback: allow configured origins
   154→  if (CLIENT_ORIGIN.includes(origin)) return callback(null, true)
   155→  logger.warn('CORS origin rejected (development)', {
   156→    module: 'cors',
   157→    origin,
   158→    allowlist: CLIENT_ORIGIN,
   159→  })
   160→  return callback(new Error(`CORS: origin ${origin} not allowed`))
   161→}

   375→// ============================================================================
   376→// Cache-Control middleware for API responses
   377→// ----------------------------------------------------------------------------
   378→// Without explicit Cache-Control, browsers apply HTTP/1.1 heuristic caching
   379→// (10% of time since Last-Modified) — which can cache sensitive data like
   380→// user profiles or chat conversations for hours.
   381→//
   382→// Rules:
   383→//   - /api/auth/*            → no-store (tokens, user data — never cache)
   384→//   - /api/chat/*            → no-store (real-time data, must be fresh)
   385→//   - /api/products*         → public, max-age=60 (catalog changes rarely)
   386→//   - /api/products/smart/*  → public, max-age=30 (smart blocks)
   387→//   - /api/stories           → public, max-age=60
   388→//   - /api/banners           → public, max-age=300 (admin-managed, slow)
   389→//   - /api/settings/*        → public, max-age=300
   390→//   - /api/push/*            → no-store (subscription endpoints are sensitive)
   391→//   - /api/calls/*           → no-store (TURN credentials per-user)
   392→//   - everything else        → no-store (default: secure)
   393→// ============================================================================
   394→app.use('/api', (req, res, next) => {
   395→  const url = req.originalUrl || req.url
   396→  // Order matters: more specific paths first
   397→  if (url.startsWith('/api/auth/') ||
   398→      url.startsWith('/api/chat/') ||
   399→      url.startsWith('/api/push/') ||
   400→      url.startsWith('/api/calls/') ||
   401→      url.startsWith('/api/leads') ||
   402→      url.startsWith('/api/orders') ||
   403→      url.startsWith('/api/upload') ||
   404→      url.startsWith('/api/favorites') ||
   405→      url.startsWith('/api/cart')) {
   406→    res.set('Cache-Control', 'no-store, no-cache, must-revalidate, private')
   407→    res.set('Pragma', 'no-cache')
   408→    res.set('Expires', '0')
   409→  } else if (url.startsWith('/api/products/smart/')) {
   410→    res.set('Cache-Control', 'public, max-age=30, stale-while-revalidate=60')
   411→  } else if (url.startsWith('/api/products')) {
   412→    res.set('Cache-Control', 'public, max-age=60, stale-while-revalidate=300')
   413→  // v12.3.1: /api/stories cache rule RESTORED (Stories module is back).
   414→  } else if (url.startsWith('/api/stories')) {
   415→    res.set('Cache-Control', 'public, max-age=60, stale-while-revalidate=300')
   416→  } else if (url.startsWith('/api/banners')) {
   417→    // v24.3: no-cache so banner changes in Studio appear immediately.
   418→    // Previously this was max-age=300 (5 min browser cache) which meant
   419→    // banner updates took up to 5 minutes to show in the app. Studio-driven
   420→    // content should never be aggressively cached.
   421→    res.set('Cache-Control', 'no-cache, must-revalidate')
   422→  } else if (url.startsWith('/api/info-pages')) {
   423→    // Public info pages — cache for 5 min, but allow stale-while-revalidate
   424→    // so updates in Studio appear quickly without forcing a full refetch.
   425→    res.set('Cache-Control', 'public, max-age=300, stale-while-revalidate=600')
   426→  } else if (url.startsWith('/api/settings/')) {
   427→    // v8-audit-fix: no-cache so changes in Studio appear immediately in the app
   428→    res.set('Cache-Control', 'no-cache, must-revalidate')
   429→  } else if (url.startsWith('/api/health') || url.startsWith('/api/ready')) {
   430→    res.set('Cache-Control', 'no-store')
   431→  } else {
   432→    // Default: secure (no-store). Override per-route if needed.
   433→    res.set('Cache-Control', 'no-store, private')
   434→  }
   435→  next()
   436→})

   549→function formatUptime(seconds: number): string {
   550→  const d = Math.floor(seconds / 86400)
   551→  const h = Math.floor((seconds % 86400) / 3600)
   552→  const m = Math.floor((seconds % 3600) / 60)
   553→  const s = Math.floor(seconds % 60)
   554→  const parts: string[] = []
   555→  if (d > 0) parts.push(`${d}d`)
   556→  if (h > 0) parts.push(`${h}h`)
   557→  if (m > 0) parts.push(`${m}m`)
   558→  parts.push(`${s}s`)
   559→  return parts.join(' ')
   560→}

   623→const shutdown = async (signal: string) => {
   624→  logger.info(`Shutdown: ${signal} received`, { module: 'shutdown' })
   625→
   626→  // v16.8-final: stop the self-destruct scheduler (cancel its interval).
   627→  stopSelfDestructScheduler()
   628→
   629→  // Stop accepting new connections
   630→  httpServer.close((err) => {
   631→    if (err) logger.error('Shutdown: HTTP close error', { module: 'shutdown', error: err })
   632→  })
   633→
   634→  // Close Socket.IO
   635→  io.close(() => {
   636→    logger.info('Shutdown: Socket.IO closed', { module: 'shutdown' })
   637→  })
   638→
   639→  // Disconnect Prisma
   640→  try {
   641→    await prisma.$disconnect()
   642→    logger.info('Shutdown: Prisma disconnected', { module: 'shutdown' })
   643→  } catch (e) {
   644→    logger.error('Shutdown: Prisma disconnect error', { module: 'shutdown', error: e instanceof Error ? e : new Error(String(e)) })
   645→  }
   646→
   647→  setTimeout(() => process.exit(0), 1000).unref()
   648→  // Hard exit if something hangs
   649→  setTimeout(() => process.exit(1), 8000).unref()
   650→}
