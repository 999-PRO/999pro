    44→    44→    44→function openPrefsDb() {
    45→    45→    45→  return new Promise((resolve, reject) => {
    46→    46→    46→    const req = indexedDB.open(NOTIF_PREFS_DB, 1)
    47→    47→    47→    req.onupgradeneeded = () => {
    48→    48→    48→      const db = req.result
    49→    49→    49→      if (!db.objectStoreNames.contains(NOTIF_PREFS_STORE)) {
    50→    50→    50→        db.createObjectStore(NOTIF_PREFS_STORE)
    51→    51→    51→      }
    52→    52→    52→    }
    53→    53→    53→    req.onsuccess = () => resolve(req.result)
    54→    54→    54→    req.onerror = () => reject(req.error)
    55→    55→    55→  })
    56→    56→    56→}

    58→    58→    58→async function refreshPrefsCache() {
    59→    59→    59→  try {
    60→    60→    60→    const db = await openPrefsDb()
    61→    61→    61→    const tx = db.transaction(NOTIF_PREFS_STORE, 'readonly')
    62→    62→    62→    const store = tx.objectStore(NOTIF_PREFS_STORE)
    63→    63→    63→    const results = await Promise.all(
    64→    64→    64→      NOTIF_PREFS_KEYS.map(
    65→    65→    65→        (key) =>
    66→    66→    66→          new Promise((resolve) => {
    67→    67→    67→            const r = store.get(key)
    68→    68→    68→            r.onsuccess = () => resolve([key, r.result === null ? true : r.result === 'true' || r.result === true])
    69→    69→    69→            r.onerror = () => resolve([key, true])
    70→    70→    70→          }),
    71→    71→    71→      ),
    72→    72→    72→    )
    73→    73→    73→    db.close()
    74→    74→    74→    const newCache = { ...prefsCache }
    75→    75→    75→    for (const [k, v] of results) newCache[k] = v
    76→    76→    76→    prefsCache = newCache
    77→    77→    77→    prefsCacheAt = Date.now()
    78→    78→    78→  } catch (e) {
    79→    79→    79→    // IDB failed — keep showing all notifications (safe default)

   749→   749→   749→function urlBase64ToUint8Array(base64String) {
   750→   750→   750→  const padding = '='.repeat((4 - (base64String.length % 4)) % 4)
   751→   751→   751→  const base64 = (base64String + padding).replace(/-/g, '+').replace(/_/g, '/')
   752→   752→   752→  const rawData = atob(base64)
   753→   753→   753→  const outputArray = new Uint8Array(rawData.length)
   754→   754→   754→  for (let i = 0; i < rawData.length; ++i) {
   755→   755→   755→    outputArray[i] = rawData.charCodeAt(i)
   756→   756→   756→  }
   757→   757→   757→  return outputArray
   758→   758→   758→}
