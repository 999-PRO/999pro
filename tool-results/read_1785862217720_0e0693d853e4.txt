    60→// ---- Recent searches (localStorage) ----
    61→const RECENTS_KEY = '999pro-universal-recents'
    62→const RECENTS_MAX = 8
    63→
    64→function loadRecents(): string[] {
    65→  try {
    66→    const raw = localStorage.getItem(RECENTS_KEY)
    67→    if (!raw) return []
    68→    const arr = JSON.parse(raw)
    69→    return Array.isArray(arr) ? arr.filter((s) => typeof s === 'string').slice(0, RECENTS_MAX) : []
    70→  } catch {
    71→    return []
    72→  }
    73→}
    74→
    75→function saveRecent(q: string) {
    76→  const trimmed = q.trim()
    77→  if (trimmed.length < 2) return
    78→  try {
    79→    const cur = loadRecents()
    80→    const next = [trimmed, ...cur.filter((s) => s.toLowerCase() !== trimmed.toLowerCase())].slice(
    81→      0,
    82→      RECENTS_MAX,
    83→    )
    84→    localStorage.setItem(RECENTS_KEY, JSON.stringify(next))
    85→  } catch {
    86→    /* ignore quota errors */
    87→  }
    88→}

   144→// ---- Highlight match helper ----
   145→function highlight(text: string, query: string): React.ReactNode {
   146→  if (!query) return text
   147→  const q = query.trim()
   148→  if (!q) return text
   149→  const lower = text.toLowerCase()
   150→  const qLower = q.toLowerCase()
   151→  if (!lower.includes(qLower)) return text
   152→  const parts: React.ReactNode[] = []
   153→  let i = 0
   154→  let key = 0
   155→  while (i < text.length) {
   156→    const idx = lower.indexOf(qLower, i)
   157→    if (idx === -1) {
   158→      parts.push(text.slice(i))
   159→      break
   160→    }
   161→    if (idx > i) parts.push(text.slice(i, idx))
   162→    parts.push(
   163→      <mark
   164→        key={key++}
   165→        style={{
   166→          background: 'rgba(139,92,246,0.3)',
   167→          color: '#fff',
   168→          borderRadius: 3,
   169→          padding: '0 2px',
   170→        }}
   171→      >
   172→        {text.slice(idx, idx + q.length)}
   173→      </mark>,
   174→    )
   175→    i = idx + q.length
   176→  }
   177→  return <>{parts}</>
   178→}
