    35→/**
    36→ * Send a Web Push notification to ALL participants of a conversation
    37→ * (except the sender).
    38→ *
    39→ * v24.5 BUGFIX: Previously only sent push to OFFLINE participants (those
    40→ * with no active socket connections). But "online" (socket connected) does
    41→ * NOT mean "looking at the app" — the tab could be in the background. In
    42→ * that case the user got the real-time socket message but NO system
    43→ * notification, so they missed messages when the tab was minimized.
    44→ *
    45→ * Now: send push to ALL participants. The Service Worker decides:
    46→ *   • If the app tab is VISIBLE → suppress the system notification (forward
    47→ *     to the page for in-app toast)
    48→ *   • If the app tab is HIDDEN or CLOSED → show a system notification
    49→ *
    50→ * This ensures the user ALWAYS gets notified, whether the app is open,
    51→ * minimized, or fully closed.
    52→ *
    53→ * Called from BOTH the socket message:send handler AND the REST POST
    54→ * /api/chat/conversations/:id/messages endpoint so the notification policy
    55→ * is consistent regardless of how the message was sent.
    56→ */
    57→export async function notifyOfflineParticipants(
    58→  conversationId: string,
    59→  senderId: string,
    60→  message: { id?: string; content?: string | null; mediaType?: string | null; attachments?: any[] | null; sender?: { displayName?: string | null; username?: string } | null },
    61→): Promise<void> {
    62→  try {
    63→    const participants = await prisma.conversationParticipant.findMany({
    64→      where: { conversationId, userId: { not: senderId } },
    65→      select: { userId: true },
    66→    })
    67→
    68→    // v24.5: send push to ALL participants (not just offline).
    69→    // The SW handles suppression when the app is visible.
    70→    const recipientIds = participants.map((p) => p.userId)
    71→
    72→    if (recipientIds.length === 0) return
    73→
    74→    const senderName = message.sender?.displayName || message.sender?.username || 'Пользователь'
    75→    // v12: use shared buildMessagePreview which handles attachments
    76→    const body = buildMessagePreview({
    77→      content: message.content,
    78→      mediaType: message.mediaType,
    79→      attachments: message.attachments,
    80→    })
    81→
    82→    // UNIQUE TAG PER MESSAGE — previously used `msg-${conversationId}` which
    83→    // caused the OS to replace previous notifications from the same
    84→    // conversation. A user receiving 5 rapid messages only saw the last
    85→    // one. Now each message has its own tag (`msg-${conversationId}-${messageId}`)
    86→    // and renotify:true so each one alerts the user. The OS shade groups
    87→    // them by conversation automatically.
    88→    const messageId = message.id || `${Date.now()}`
    89→    const tag = `msg-${conversationId}-${messageId}`
    90→
    91→    for (const recipientId of recipientIds) {
    92→      void sendPushToUser(recipientId, {
    93→        title: senderName,
    94→        body,
    95→        tag,
    96→        url: '/?view=chat',
    97→        data: { conversationId, messageId },
    98→        renotify: true,
    99→      }).catch((e) => {
   100→        // Log push failures so we can diagnose "user didn't get notified" issues.
   101→        // Previously this was silently swallowed — made debugging impossible.
   102→        console.error('[PUSH] notifyOfflineParticipants: sendPushToUser failed for', recipientId, e instanceof Error ? e.message : e)
   103→      })
   104→    }
   105→  } catch (e) {
   106→    // Push is best-effort — never fail the message send because of a push error.
   107→    // But log it so we can diagnose "user didn't get notified" issues.
   108→    console.error('[PUSH] notifyOfflineParticipants error:', e instanceof Error ? e.message : e)
   109→  }
   110→}

   147→async function tryAuth(socket: Socket): Promise<{ userId: string; username: string } | null> {
   148→  const token =
   149→    (socket.handshake.auth?.token as string | undefined) ||
   150→    (socket.handshake.headers.authorization?.replace(/^Bearer\s+/i, '') as string | undefined)
   151→  if (!token) return null
   152→  try {
   153→    const decoded: JwtPayload = verifyToken(token)
   154→    // B-HIGH-008: short-circuit the DB lookup when the user's auth entry is
   155→    // already cached by a recent REST request. Without this, every socket
   156→    // connect (initial, reconnect, multiple tabs, mobile background→foreground
   157→    // transitions) triggered a `prisma.user.findUnique` — under load this
   158→    // produced thousands of identical queries per second on the same user ids
   159→    // that had JUST been validated by requireAuth on the bootstrap HTTP
   160→    // request. The cache invalidates automatically on token-version bump
   161→    // (getCachedAuth compares v and deletes on mismatch).
   162→    const cached = getCachedAuth(decoded.sub, decoded.v)
   163→    if (cached) {
   164→      return { userId: decoded.sub, username: decoded.username }
   165→    }
   166→    // Cache miss — fall through to DB lookup, then populate the cache so the
   167→    // NEXT socket connect (and any REST request) hits the cache.
   168→    const user = await prisma.user.findUnique({
   169→      where: { id: decoded.sub },
   170→      select: { tokenVersion: true, deletedAt: true, role: true },
   171→    })
   172→    if (!user || user.deletedAt) return null
   173→    if (user.tokenVersion !== decoded.v) return null
   174→    setCachedAuth(decoded.sub, {
   175→      v: user.tokenVersion,
   176→      role: user.role as 'user' | 'admin',
   177→      expires: Date.now() + 60_000,
   178→    })
   179→    return { userId: decoded.sub, username: decoded.username }
   180→  } catch {
   181→    return null
   182→  }
   183→}

   185→/**
   186→ * Per-socket rate limiter — token bucket per event name.
   187→ * Limits how fast a single client can emit any given event.
   188→ *
   189→ * Without this, a single malicious user can flood message:send or
   190→ * typing:start at thousands of events per second, each triggering a DB
   191→ * write + push dispatch + room broadcast.
   192→ */
   193→interface RateBucket { tokens: number; lastRefill: number }
   194→const RATE_LIMITS: Record<string, { capacity: number; refillPerSec: number }> = {
   195→  'message:send': { capacity: 8, refillPerSec: 2 }, // 8 burst, then 2/sec
   196→  'message:forward': { capacity: 5, refillPerSec: 1 },
   197→  'typing:start': { capacity: 5, refillPerSec: 2 },
   198→  'message:read': { capacity: 20, refillPerSec: 5 },
   199→  'conversation:join': { capacity: 30, refillPerSec: 10 },
   200→}
   201→
   202→/**
   203→ * Force-disconnect every socket owned by `userId`. Call this when the
   204→ * user's tokenVersion is bumped (password change, role demotion, remote
   205→ * logout) so that they have to re-authenticate to keep using real-time
   206→ * channels.
   207→ *
   208→ * IMPORTANT: We do NOT clear `userSockets.get(userId)` here. The disconnect
   209→ * is async — the socket's `disconnect` event fires later, and the
   210→ * `disconnect` handler (below) is what cleans up `sockets.delete(socket.id)`
   211→ * + calls `setUserOffline` when the set becomes empty. If we cleared here,
   212→ * the disconnect handler would find an empty Set / undefined map and skip
   213→ * the offline transition — leaving isOnline=true in the DB forever.
   214→ */
   215→export function kickUserSockets(userId: string) {
   216→  const sockets = userSockets.get(userId)
   217→  if (!sockets) return
   218→  // Snapshot to array so we don't mutate during iteration.
   219→  const snapshot = Array.from(sockets)
   220→  for (const socketId of snapshot) {
   221→    const socket = ioInstance?.sockets.sockets.get(socketId)
   222→    if (socket) socket.disconnect(true)
   223→  }
   224→}

   226→function rateLimit(socket: Socket, event: string): boolean {
   227→  // H4 fix: FAIL-CLOSED для unmapped events. Раньше возвращалось `true`
   228→  // (разрешить), что означало: любое новое событие, добавленное без записи в
   229→  // RATE_LIMITS, полностью обходило лимитирование (DoS-поверхность). Теперь
   230→  // применяем консервативный default-лимит (1 запрос в секунду, burst 3) —
   231→  // достаточно для легитимных редких событий, но блокирует flood.
   232→  const cfg = RATE_LIMITS[event] ?? { capacity: 3, refillPerSec: 1 }
   233→  const map = (socket.data.rateBuckets ??= new Map<string, RateBucket>()) as Map<string, RateBucket>
   234→  let bucket = map.get(event)
   235→  const now = Date.now()
   236→  if (!bucket) {
   237→    bucket = { tokens: cfg.capacity, lastRefill: now }
   238→    map.set(event, bucket)
   239→  } else {
   240→    const elapsed = (now - bucket.lastRefill) / 1000
   241→    bucket.tokens = Math.min(cfg.capacity, bucket.tokens + elapsed * cfg.refillPerSec)
   242→    bucket.lastRefill = now
   243→  }
   244→  if (bucket.tokens < 1) return false
   245→  bucket.tokens -= 1
   246→  return true
   247→}

   255→async function setUserOffline(userId: string) {

   259→    .update({ where: { id: userId }, data: { isOnline: false, lastSeen: new Date() } })

   294→// This allows REST route handlers (e.g. POST /api/chat/conversations/:id/messages)
   295→// to broadcast `message:received` events so messages sent via REST API reach

   299→export function getIo(): IoServer | null {
   300→  return ioInstance

   303→export function registerChatHandlers(io: IoServer) {
   304→  ioInstance = io
   305→  io.use(async (socket, next) => {
   306→    const auth = await tryAuth(socket)
   307→    if (!auth) return next(new Error('Unauthorized'))
   308→    ;(socket.data as any).auth = auth
   309→    next()
   310→  })
   311→
   312→  io.on('connection', async (socket: Socket) => {
   313→    const auth = (socket.data as any).auth as { userId: string; username: string }
   314→    if (!auth) {
   315→      socket.disconnect(true)
   316→      return
   317→    }
   318→
   319→    const { userId, username } = auth
   320→
   321→    // SECURITY: cap concurrent sockets per user to prevent DoS. A malicious
   322→    // or buggy client can otherwise open thousands of sockets per second,
   323→    // each triggering setUserOnline + presence broadcast + auto-join loop.
   324→    // 10 is enough for any legitimate use (multiple tabs + multiple devices).
   325→    const MAX_SOCKETS_PER_USER = 10
   326→    const existingSockets = userSockets.get(userId)
   327→    if (existingSockets && existingSockets.size >= MAX_SOCKETS_PER_USER) {
   328→      console.warn(`[SOCKET] Rejecting socket for ${username}: ${existingSockets.size} sockets already open (cap=${MAX_SOCKETS_PER_USER})`)
   329→      socket.emit('error', { message: 'Too many concurrent connections' })
   330→      socket.disconnect(true)
   331→      return
   332→    }
   333→
   334→    socketToUser.set(socket.id, { userId, username })
   335→    let sockets = userSockets.get(userId)
   336→    if (!sockets) {
   337→      sockets = new Set()
   338→      userSockets.set(userId, sockets)
   339→    }
   340→    sockets.add(socket.id)
   341→    void setUserOnline(userId)
   342→    void broadcastPresenceToPartners(io, userId, 'user:online', { userId, username })
   343→
   344→    // ========================================================================
   345→    // PERSONAL + ADMIN ROOMS
   346→    // Every socket joins `user:<userId>` so we can target events (like
   347→    // lead:status-changed) directly at the user regardless of which
   348→    // conversation they're viewing. If the user is an admin, they also
   349→    // join the `admins` room — used for admin-wide broadcasts such as
   350→    // lead status updates visible to all admin studios.
   351→    // ========================================================================
   352→    socket.join(`user:${userId}`)
   353→    try {
   354→      const u = await prisma.user.findUnique({
   355→        where: { id: userId },
   356→        select: { role: true },
   357→      })
   358→      if (u?.role === 'admin') {
   359→        socket.join('admins')
   360→      }
   361→    } catch {
   362→      /* non-critical */
   363→    }
   364→
   365→    // ========================================================================
   366→    // AUTO-JOIN all of the user's conversations.
   367→    // This is the critical fix: previously the frontend only joined a room
   368→    // when the user opened a specific conversation. That meant messages in
   369→    // other conversations were NEVER delivered to the user when they were
   370→    // not viewing that chat (or were elsewhere in the app). Now the server
   371→    // joins every connected socket to every conversation the user is a
   372→    // participant of, so `message:received` broadcasts reach them no matter
   373→    // which view they're on. The client-side `conversation:join` is kept as
   374→    // a no-op compatibility shim.
   375→    //
   376→    // v13.1 (audit P1-12 fix): emit a 'ready' event after the auto-join
   377→    // completes so the client can wait for it before sending messages.
   378→    // Previously the IIFE was fire-and-forget — between socket connection
   379→    // and the IIFE completing, any `message:received` broadcast to a
   380→    // conversation room would NOT reach this socket. The client doesn't
   381→    // strictly need to wait on 'ready' (older clients still work), but
   382→    // new clients CAN wait to guarantee delivery.
   383→    ;(async () => {
   384→      try {
   385→        const participations = await prisma.conversationParticipant.findMany({
   386→          where: { userId },
   387→          select: { conversationId: true },
   388→        })
   389→        for (const p of participations) {
   390→          socket.join(`conversation:${p.conversationId}`)
   391→        }
   392→        socket.data.ready = true
   393→        socket.emit('ready', { userId })
   394→      } catch (e) {
   395→        console.error('[SOCKET] auto-join conversations error:', e)
   396→      }
   397→    })()
   398→
   399→    // ====== Conversation room management (kept for backwards compat) ======
   400→    socket.on('conversation:join', async (conversationId: string) => {
   401→      try {
   402→        if (!conversationId || typeof conversationId !== 'string') return
   403→        const ok = await isParticipant(conversationId, userId)
   404→        if (!ok) {
   405→          socket.emit('error', { event: 'conversation:join', error: 'Not a participant' })
   406→          return
   407→        }
   408→        socket.join(`conversation:${conversationId}`)
   409→      } catch (e) {
   410→        socket.emit('error', { event: 'conversation:join', error: 'Internal error' })
   411→      }
   412→    })
   413→
   414→    // conversation:leave is now a NO-OP — the server auto-joins every
   415→    // socket to every conversation the user is a participant of. Leaving a
   416→    // room would silently break `message:received` delivery for that
   417→    // conversation, killing in-app toast notifications, unread-badge
   418→    // increments, and chat-list preview refreshes for any conversation
   419→    // the user has ever opened. Per-conversation filtering is done
   420→    // client-side in chat.tsx.
   421→    //
   422→    // The handler is kept (not deleted) so legacy clients that still emit
   423→    // the event don't get an "unknown event" warning in the socket.io logs.
   424→    socket.on('conversation:leave', (_conversationId: string) => {
   425→      // intentionally no-op
   426→    })
   427→
   428→    // ====== Typing indicators ======
   429→    // Server-side TTL: if a client sends `typing:start` and then crashes /
   430→    // closes the tab before sending `typing:stop`, other participants would
   431→    // see a permanent "печатает…" indicator. We arm a 5-second timer on each
   432→    // `typing:start` — if the client doesn't refresh or explicitly stop, we
   433→    // auto-broadcast `typing:stop` to the room.
   434→    // Map keyed by `${conversationId}:${userId}` so multiple conversations
   435→    // from the same user (impossible in 1:1, but possible in group) don't
   436→    // collide. `timer.unref()` prevents the timer from keeping the event
   437→    // loop alive on shutdown.
   438→    const typingTimerKey = (conversationId: string) => `${conversationId}:${userId}`
   439→    const clearTypingTimer = (conversationId: string) => {
   440→      const t = typingTimers.get(typingTimerKey(conversationId))
   441→      if (t) {
   442→        clearTimeout(t)
   443→        typingTimers.delete(typingTimerKey(conversationId))
   444→      }
   445→    }
   446→
   447→    socket.on('typing:start', async ({ conversationId }: { conversationId: string }) => {
   448→      try {
   449→        if (!conversationId) return
   450→        if (!rateLimit(socket, 'typing:start')) return
   451→        if (!(await isParticipant(conversationId, userId))) return
   452→        socket.to(`conversation:${conversationId}`).emit('typing:start', { conversationId, userId, username })
   453→
   454→        // (Re)arm the TTL timer. If an existing timer is already running
   455→        // (client refreshed `typing:start`), clear it first to extend the
   456→        // window.
   457→        clearTypingTimer(conversationId)
   458→        const timer = setTimeout(() => {
   459→          socket.to(`conversation:${conversationId}`).emit('typing:stop', { conversationId, userId })
   460→          typingTimers.delete(typingTimerKey(conversationId))
   461→        }, 5000)
   462→        timer.unref?.()
   463→        typingTimers.set(typingTimerKey(conversationId), timer)
   464→      } catch {}
   465→    })
   466→
   467→    socket.on('typing:stop', async ({ conversationId }: { conversationId: string }) => {
   468→      try {
   469→        if (!conversationId) return
   470→        if (!(await isParticipant(conversationId, userId))) return
   471→        socket.to(`conversation:${conversationId}`).emit('typing:stop', { conversationId, userId })
   472→        clearTypingTimer(conversationId)
   473→      } catch {}
   474→    })
   475→
   476→    // ====== Voice recording indicators (v9-voice) ======
   477→    // When a user starts recording a voice message, broadcast to the room
   478→    // so other participants see "username записывает голосовое сообщение…"
   479→    // Same pattern as typing:start/stop — no persistence, no DB writes.
   480→    socket.on('voice:recording:start', async ({ conversationId }: { conversationId: string }) => {
   481→      try {
   482→        if (!conversationId) return
   483→        if (!rateLimit(socket, 'typing:start')) return // reuse typing rate limit
   484→        if (!(await isParticipant(conversationId, userId))) return
   485→        socket.to(`conversation:${conversationId}`).emit('voice:recording:start', { conversationId, userId, username })
   486→      } catch {}
   487→    })
   488→
   489→    socket.on('voice:recording:stop', async ({ conversationId }: { conversationId: string }) => {
   490→      try {
   491→        if (!conversationId) return
   492→        if (!(await isParticipant(conversationId, userId))) return
   493→        socket.to(`conversation:${conversationId}`).emit('voice:recording:stop', { conversationId, userId })
   494→      } catch {}
   495→    })
   496→
   497→    // ====== Send message (with reply / forward support) ======
   498→    socket.on(
   499→      'message:send',
   500→      async (payload: {
   501→        conversationId: string
   502→        content?: string
   503→        mediaUrl?: string
   504→        mediaType?: 'text' | 'image' | 'video' | 'audio' | 'file' | 'product' | 'audio-hub' | 'film'
   505→        duration?: number
   506→        tempId?: string
   507→        replyToId?: string
   508→        forwardedFromId?: string
   509→        // v12-fix: attachments array for multi-file messages. Was missing
   510→        // from the payload type (runtime worked, but TS complained).
   511→        attachments?: Array<{ url: string; type: string; name?: string; size?: number; duration?: number }>
   512→        // v16.8-final: optional self-destruct timer (voice messages).
   513→        // Validated by sendMessageSchema to be one of 0, 60, 720, 1440, 10080.
   514→        selfDestructMinutes?: number
   515→      }) => {
   516→        try {
   517→          if (!rateLimit(socket, 'message:send')) {
   518→            socket.emit('error', { event: 'message:send', error: 'Rate limit exceeded' })
   519→            return
   520→          }
   521→          if (!payload?.conversationId) {
   522→            socket.emit('error', { event: 'message:send', error: 'conversationId required' })
   523→            return
   524→          }
   525→          const data = sendMessageSchema.parse({
   526→            content: payload.content,
   527→            mediaUrl: payload.mediaUrl,
   528→            mediaType: payload.mediaType,
   529→            // v12-fix: pass attachments to schema validation (was missing →
   530→            // multi-file messages lost their attachments on socket send path)
   531→            attachments: payload.attachments,
   532→            duration: payload.duration,
   533→            replyToId: payload.replyToId,
   534→            forwardedFromId: payload.forwardedFromId,
   535→            selfDestructMinutes: payload.selfDestructMinutes,
   536→          })
   537→
   538→          if (!(await isParticipant(payload.conversationId, userId))) {
   539→            socket.emit('error', { event: 'message:send', error: 'Not a participant' })
   540→            return
   541→          }
   542→
   543→          // Validate reply target if provided
   544→          if (data.replyToId) {
   545→            const replied = await prisma.message.findUnique({
   546→              where: { id: data.replyToId },
   547→              select: { conversationId: true },
   548→            })
   549→            if (!replied || replied.conversationId !== payload.conversationId) {
   550→              socket.emit('error', { event: 'message:send', error: 'Reply target not in this conversation' })
   551→              return
   552→            }
   553→          }
   554→
   555→          // C-CRIT-002 fix: idempotency via clientMessageId. If the client
   556→          // provides a tempId, check if a message with the same clientMessageId
   557→          // already exists (e.g. socket retry, REST fallback after socket
   558→          // timeout). If so, return the existing message instead of creating
   559→          // a duplicate. This prevents the "duplicate messages on flaky
   560→          // mobile network" bug.
   561→          const clientMessageId = payload.tempId || null
   562→          if (clientMessageId) {
   563→            const existing = await prisma.message.findFirst({
   564→              where: { senderId: userId, clientMessageId },
   565→              include: {
   566→                sender: { select: userSelect },
   567→                replyTo: { include: { sender: { select: userSelect } } },
   568→                forwardedFrom: { include: { sender: { select: userSelect } } },
   569→              },
   570→            })
   571→            if (existing) {
   572→              // Idempotent response — return the existing message with tempId
   573→              // tag so the client reconciles its optimistic UI.
   574→              const formatted = serialiseMessage(existing, '')
   575→              ;(formatted as any).tempId = clientMessageId
   576→              socket.emit('message:received', formatted)
   577→              return
   578→            }
   579→          }
   580→
   581→          // v16.9 MODERATION — run two-level check before persisting.
   582→          // If blocked, emit 'message:blocked' to the sender with the reason
   583→          // and DO NOT persist the message.
   584→          if (data.content && data.content.trim()) {
   585→            const modDecision = await moderateContent(data.content, {
   586→              userId,
   587→              targetType: 'message',
   588→              conversationId: payload.conversationId,
   589→            })
   590→            if (!modDecision.allowed) {
   591→              socket.emit('message:blocked', {
   592→                tempId: payload.tempId,
   593→                reason: modDecision.reason,
   594→                severity: modDecision.severity,
   595→                categories: modDecision.categories,
   596→              })
   597→              return
   598→            }
   599→          }
   600→
   601→          const message = await prisma.message.create({
   602→            data: {
   603→              conversationId: payload.conversationId,
   604→              senderId: userId,
   605→              content: data.content,
   606→              mediaUrl: data.mediaUrl,
   607→              mediaType: data.mediaType,
   608→              // v12: store attachments as JSON string if present
   609→              attachments: data.attachments ? JSON.stringify(data.attachments) : null,
   610→              duration: data.duration,
   611→              replyToId: data.replyToId,
   612→              forwardedFromId: data.forwardedFromId,
   613→              clientMessageId, // C-CRIT-002: persist for future dedup
   614→              // v16.8-final: self-destruct timer. If selfDestructMinutes > 0
   615→              // and the message is a voice message, compute the deadline from
   616→              // now. 0 / undefined → null (no auto-delete).
   617→              selfDestructAt:
   618→                data.selfDestructMinutes && data.selfDestructMinutes > 0
   619→                  ? new Date(Date.now() + data.selfDestructMinutes * 60 * 1000)
   620→                  : null,
   621→            },
   622→            include: {
   623→              sender: { select: userSelect },
   624→              replyTo: { include: { sender: { select: userSelect } } },
   625→              forwardedFrom: { include: { sender: { select: userSelect } } },
   626→            },
   627→          })
   628→
   629→          await prisma.conversation.update({
   630→            where: { id: payload.conversationId },
   631→            data: { updatedAt: new Date() },
   632→          })
   633→
   634→          // C-HIGH-001 fix: tempId is sender-side reconciliation key — broadcast
   635→          // only to the sender's socket. Other participants receive the message
   636→          // without tempId (prevents minor info leak + dedup confusion).
   637→          const senderFormatted = serialiseMessage(message, '')
   638→          ;(senderFormatted as any).tempId = payload.tempId
   639→          socket.emit('message:received', senderFormatted)
   640→          const othersFormatted = serialiseMessage(message, '')
   641→          socket.to(`conversation:${payload.conversationId}`).emit('message:received', othersFormatted)
   642→
   643→          // ====================================================================
   644→          // WEB PUSH NOTIFICATION — delegate to the shared helper so the
   645→          // logic stays in sync with the REST endpoint. Previously this was
   646→          // duplicated inline and would drift from notifyOfflineParticipants.
   647→          // ====================================================================
   648→          void notifyOfflineParticipants(payload.conversationId, userId, {
   649→            id: message.id,
   650→            content: message.content,
   651→            mediaType: message.mediaType,
   652→            sender: message.sender,
   653→          })
   654→        } catch (e: any) {
   655→          socket.emit('error', {
   656→            event: 'message:send',
   657→            error: e?.issues?.[0]?.message || e?.message || 'Failed to send message',
   658→          })
   659→        }
   660→      },
   661→    )

   682→    socket.on(
   683→      'message:delete',
   684→      async (payload: { messageId: string; conversationId: string; forEveryone: boolean }) => {
   685→        try {
   686→          if (!payload?.messageId || !payload?.conversationId) return
   687→          if (!(await isParticipant(payload.conversationId, userId))) {
   688→            socket.emit('error', { event: 'message:delete', error: 'Not a participant' })
   689→            return
   690→          }
   691→
   692→          const msg = await prisma.message.findUnique({
   693→            where: { id: payload.messageId },
   694→            select: { id: true, senderId: true, deletedFor: true, deletedForAll: true, conversationId: true },
   695→          })
   696→          if (!msg || msg.conversationId !== payload.conversationId) {
   697→            socket.emit('error', { event: 'message:delete', error: 'Message not found' })
   698→            return
   699→          }
   700→
   701→          const isOwner = msg.senderId === userId
   702→
   703→          if (payload.forEveryone) {
   704→            if (!isOwner) {
   705→              socket.emit('error', { event: 'message:delete', error: 'Only sender can delete for everyone' })
   706→              return
   707→            }
   708→            await prisma.message.update({
   709→              where: { id: msg.id },
   710→              data: { deletedForAll: true, content: null, mediaUrl: null, mediaType: null, duration: null },
   711→            })

   713→              messageId: msg.id,
   714→              conversationId: payload.conversationId,
   715→              deletedForAll: true,
   716→            })
   717→          } else {
   718→            const deletedFor: string[] = (() => {
   719→              try { return JSON.parse(msg.deletedFor || '[]') } catch { return [] }
   720→            })()
   721→            if (!deletedFor.includes(userId)) {
   722→              deletedFor.push(userId)
   723→              await prisma.message.update({
   724→                where: { id: msg.id },
   725→                data: { deletedFor: JSON.stringify(deletedFor) },
   726→              })
   727→            }
   728→            // Only inform the user's own sockets — others should not see this
   729→            emitToUserSockets(io, userId, 'message:deleted', {
   730→              messageId: msg.id,
   731→              conversationId: payload.conversationId,
   732→              deletedForMe: true,
   733→            })
   734→          }
   735→        } catch (e: any) {
   736→          socket.emit('error', { event: 'message:delete', error: e?.message || 'Failed to delete' })
   737→        }
   738→      },
   739→    )

   743→      'message:forward',
   744→      async (payload: { sourceMessageId: string; targetConversationIds: string[] }) => {
   745→        try {
   746→          if (!payload?.sourceMessageId || !Array.isArray(payload.targetConversationIds)) {
   747→            socket.emit('error', { event: 'message:forward', error: 'Invalid payload' })
   748→            return
   749→          }
   750→          if (payload.targetConversationIds.length === 0 || payload.targetConversationIds.length > 20) {
   751→            socket.emit('error', { event: 'message:forward', error: 'Target conversations count out of range' })
   752→            return
   753→          }
   754→
   755→          const source = await prisma.message.findUnique({
   756→            where: { id: payload.sourceMessageId },
   757→          })
   758→          if (!source || source.deletedForAll) {
   759→            socket.emit('error', { event: 'message:forward', error: 'Source message not available' })
   760→            return
   761→          }
   762→
   763→          // C-CRIT-001 fix: verify the caller is a participant of the SOURCE
   764→          // conversation. Without this check, any authenticated user could
   765→          // forward ANY message by ID (privacy breach — they could exfiltrate
   766→          // message content + mediaUrl from any conversation in the system
   767→          // by guessing/enumerating Prisma cuids).
   768→          // The REST endpoint POST /api/chat/messages/:id/forward already has
   769→          // this check (SECURITY FIX comment at routes/chat.ts:580-606), but
   770→          // the socket handler was missed.
   771→          if (!(await isParticipant(source.conversationId, userId))) {
   772→            socket.emit('error', { event: 'message:forward', error: 'Source message not available' })
   773→            return
   774→          }

   777→            if (!(await isParticipant(targetId, userId))) continue
   778→

   780→              data: {
   781→                conversationId: targetId,
   782→                senderId: userId,

   787→                forwardedFromId: source.id,
   788→              },
   789→              include: {
   790→                sender: { select: userSelect },
   791→                replyTo: { include: { sender: { select: userSelect } } },
   792→                forwardedFrom: { include: { sender: { select: userSelect } } },
   793→              },
   794→            })
   795→            await prisma.conversation.update({
   796→              where: { id: targetId },
   797→              data: { updatedAt: new Date() },
   798→            })
   799→            const formatted = serialiseMessage(msg, '')
   800→            io.to(`conversation:${targetId}`).emit('message:received', formatted)
   801→          }
   802→          socket.emit('message:forwarded', { sourceMessageId: payload.sourceMessageId, count: payload.targetConversationIds.length })
   803→        } catch (e: any) {
   804→          socket.emit('error', { event: 'message:forward', error: e?.message || 'Forward failed' })
   805→        }
   806→      },

   815→    // Initiate a call — creates a Call record and notifies the recipient
   816→    socket.on(
   817→      'call:start',
   818→      async (payload: { conversationId: string; recipientId: string; type: 'audio' | 'video' }) => {
   819→        try {
   820→          if (!payload?.conversationId || !payload?.recipientId || !payload?.type) {
   821→            socket.emit('error', { event: 'call:start', error: 'Invalid payload' })
   822→            return
   823→          }
   824→          if (!(await isParticipant(payload.conversationId, userId))) {
   825→            socket.emit('error', { event: 'call:start', error: 'Not a participant' })
   826→            return
   827→          }
   828→          if (payload.recipientId === userId) {
   829→            socket.emit('error', { event: 'call:start', error: 'Cannot call yourself' })
   830→            return
   831→          }
   832→
   833→          // Don't allow calling an offline user
   834→          const recipient = await prisma.user.findUnique({
   835→            where: { id: payload.recipientId },
   836→            select: { id: true, isOnline: true, username: true, displayName: true, avatar: true },
   837→          })
   838→          if (!recipient) {
   839→            socket.emit('error', { event: 'call:start', error: 'Recipient not found' })
   840→            return
   841→          }
   842→          if (!recipient.isOnline && !userSockets.has(recipient.id)) {
   843→            socket.emit('error', { event: 'call:start', error: 'User is offline' })
   844→            return
   845→          }
   846→
   847→          const call = await prisma.call.create({
   848→            data: {
   849→              conversationId: payload.conversationId,
   850→              callerId: userId,
   851→              recipientId: payload.recipientId,
   852→              type: payload.type,
   853→              status: 'ringing',
   854→            },
   855→          })
   856→
   857→          const caller = await prisma.user.findUnique({
   858→            where: { id: userId },
   859→            select: userSelect,
   860→          })
   861→
   862→          // Notify the recipient (across all their sockets — multi-tab)
   863→          emitToUserSockets(io, payload.recipientId, 'call:incoming', {
   864→            callId: call.id,
   865→            conversationId: payload.conversationId,
   866→            type: payload.type,
   867→            caller: caller
   868→              ? {
   869→                  id: caller.id,
   870→                  username: caller.username,
   871→                  displayName: caller.displayName,
   872→                  avatar: caller.avatar,
   873→                }
   874→              : null,
   875→          })
   876→          socket.emit('call:started', { callId: call.id, conversationId: payload.conversationId })
   877→
   878→          // ====== MISSED CALL TIMEOUT ======
   879→          // If the recipient doesn't answer within 45 seconds, auto-mark the
   880→          // call as 'missed' and notify both parties. Without this, the call
   881→          // stays in 'ringing' forever — the caller sees "Вызов…" endlessly
   882→          // and the recipient's phone keeps vibrating if push was sent.
   883→          //
   884→          // 45s is the industry standard (Telegram, WhatsApp, Viber all use
   885→          // 30-60s). Long enough for the recipient to grab their phone, short
   886→          // enough to not waste the caller's time.
   887→          setTimeout(() => {
   888→            (async () => {
   889→              try {
   890→                const fresh = await prisma.call.findUnique({
   891→                  where: { id: call.id },
   892→                  select: { id: true, status: true, callerId: true, recipientId: true },
   893→                })
   894→                if (!fresh) return
   895→                if (fresh.status !== 'ringing') return // already accepted/rejected/cancelled
   896→                await prisma.call.update({
   897→                  where: { id: call.id },
   898→                  data: { status: 'missed', endedAt: new Date() },
   899→                })
   900→                // Notify both: caller sees "missed", recipient sees "missed" too
   901→                emitToUserSockets(io, fresh.callerId, 'call:ended', {
   902→                  callId: call.id,
   903→                  reason: 'missed',
   904→                  duration: null,
   905→                })
   906→                emitToUserSockets(io, fresh.recipientId, 'call:ended', {
   907→                  callId: call.id,
   908→                  reason: 'missed',
   909→                  duration: null,
   910→                })
   911→              } catch (e) {
   912→                console.error('[CALL] missed timeout error:', e instanceof Error ? e.message : e)
   913→              }
   914→            })()
   915→          }, 45_000).unref?.() // unref so the timer doesn't keep Node alive on shutdown
   916→
   917→          // WEB PUSH for incoming call — fires when the recipient has NO
   918→          // active socket connection (app fully closed or backgrounded).
   919→          // The push notification includes Answer/Decline action buttons
   920→          // that the SW handles via `notificationclick` -> event.action.
   921→          // requireInteraction: true so the notification persists until
   922→          // the user acts on it (otherwise Android auto-dismisses after
   923→          // ~5s and the user misses the call).
   924→          if (!isUserOnline(payload.recipientId)) {
   925→            const callerName = caller?.displayName || caller?.username || 'Пользователь'
   926→            const callTitle = `Входящий ${payload.type === 'video' ? 'видеозвонок' : 'аудиозвонок'}`
   927→            void sendPushToUser(payload.recipientId, {
   928→              title: callTitle,
   929→              body: `От ${callerName}`,
   930→              tag: `call-${call.id}`,
   931→              url: '/?view=chat',
   932→              data: {
   933→                conversationId: payload.conversationId,
   934→                callId: call.id,
   935→                callType: payload.type,
   936→                callerId: userId,
   937→              },
   938→              requireInteraction: true,
   939→              renotify: false, // call tag is already unique per call
   940→              actions: [
   941→                { action: 'answer', title: 'Ответить' },
   942→                { action: 'decline', title: 'Отклонить' },
   943→              ],
   944→            })
   945→          }
   946→        } catch (e: any) {
   947→          socket.emit('error', { event: 'call:start', error: e?.message || 'Call start failed' })
   948→        }
   949→      },
   950→    )

  1005→    socket.on('call:end', async (payload: { callId: string; reason?: string }) => {
  1006→      try {
  1007→        if (!payload?.callId) return
  1008→        const call = await prisma.call.findUnique({
  1009→          where: { id: payload.callId },
  1010→          select: { id: true, callerId: true, recipientId: true, status: true, startedAt: true },

  1020→          data: { status: 'ended', endedAt: new Date(), duration },
  1021→        })
  1022→        emitToUserSockets(io, call.callerId, 'call:ended', { callId: call.id, reason: payload.reason || 'ended', duration })
  1023→        emitToUserSockets(io, call.recipientId, 'call:ended', { callId: call.id, reason: payload.reason || 'ended', duration })
  1024→      } catch (e: any) {
  1025→        socket.emit('error', { event: 'call:end', error: e?.message })
  1026→      }
  1027→    })
  1028→
  1029→    // Cancel an outgoing call before the recipient answers
  1030→    socket.on('call:cancel', async (payload: { callId: string }) => {
  1031→      try {
  1032→        if (!payload?.callId) return
  1033→        const call = await prisma.call.findUnique({
  1034→          where: { id: payload.callId },
  1035→          select: { id: true, callerId: true, recipientId: true, status: true },
  1036→        })
  1037→        if (!call) return
  1038→        if (call.callerId !== userId) return
  1039→        if (call.status !== 'ringing') return
  1040→        await prisma.call.update({
  1041→          where: { id: call.id },
  1042→          data: { status: 'cancelled', endedAt: new Date() },
  1043→        })
  1044→        emitToUserSockets(io, call.recipientId, 'call:cancelled', { callId: call.id })
  1045→      } catch (e: any) {
  1046→        socket.emit('error', { event: 'call:cancel', error: e?.message })
  1047→      }
  1048→    })
  1049→
  1050→    // WebRTC signaling — relay offer/answer/ICE candidates to the peer.
  1051→    // The server does NOT touch the media; it just forwards signaling messages.
  1052→    socket.on('call:signal', async (payload: { callId: string; to: string; data: any }) => {
  1053→      try {
  1054→        if (!payload?.callId || !payload?.to || !payload?.data) return
  1055→        // B-HIGH-007: cap relayed signal payload to 16 KiB. Without this, a
  1056→        // malicious or buggy client can ship an arbitrarily-large `data`
  1057→        // object (e.g. SDP with embedded data channels, base64 media) which
  1058→        // the server happily re-serialises and broadcasts to every socket of
  1059→        // the peer — a trivial amplification / memory-exhaustion vector.
  1060→        // 16 KiB is far above any legit SDP+ICE bundle (typical: 2–4 KiB).
  1061→        const dataStr = JSON.stringify(payload.data)
  1062→        if (dataStr.length > 16_384) {
  1063→          socket.emit('error', { event: 'call:signal', error: 'Signal payload too large' })
  1064→          return
  1065→        }
  1066→        // Verify the caller is part of this call
  1067→        const call = await prisma.call.findUnique({
  1068→          where: { id: payload.callId },
  1069→          select: { callerId: true, recipientId: true, status: true },
  1070→        })
  1071→        if (!call) return
  1072→        if (call.callerId !== userId && call.recipientId !== userId) return
  1073→        // v13.0 (audit P0-3 fix): the previous version only validated that
  1074→        // the SENDER (userId) is a participant, but did not validate that
  1075→        // payload.to (the recipient of the relayed signaling) is the OTHER
  1076→        // participant. An attacker in a call with user B could relay SDP/ICE
  1077→        // to ANY arbitrary user ID — leaking their ICE candidates (IPs)
  1078→        // and spamming their UI. Now strictly require payload.to to be the
  1079→        // other party of the same call.
  1080→        if (payload.to !== call.callerId && payload.to !== call.recipientId) return
  1081→        if (payload.to === userId) return // cannot signal yourself
  1082→        // v13.0 (audit P1-11 fix): only relay signaling once the call is
  1083→        // accepted. Previously `ringing` was allowed — letting the caller
  1084→        // push SDP/ICE to the recipient before they consented to the call,
  1085→        // leaking their ICE candidates. Now strictly require `accepted`.
  1086→        if (call.status !== 'accepted') return
  1087→        // Forward to the peer's sockets
  1088→        emitToUserSockets(io, payload.to, 'call:signal', {
  1089→          callId: payload.callId,
  1090→          from: userId,
  1091→          data: payload.data,
  1092→        })
  1093→      } catch (e: any) {
  1094→        socket.emit('error', { event: 'call:signal', error: e?.message })
  1095→      }
  1096→    })
  1097→
  1098→    // ====== Disconnect ======
  1099→    socket.on('disconnect', async () => {
  1100→      socketToUser.delete(socket.id)
  1101→      const sockets = userSockets.get(userId)
  1102→      if (sockets) {
  1103→        sockets.delete(socket.id)
  1104→        if (sockets.size === 0) {
  1105→          userSockets.delete(userId)
  1106→          await setUserOffline(userId)
  1107→          void broadcastPresenceToPartners(io, userId, 'user:offline', { userId, username })
  1108→
  1109→          // ====== AUTO-END ACTIVE CALLS ON DISCONNECT ======
  1110→          // If the user had an active call and their socket dropped without
  1111→          // calling call:end, we need to clean up. Otherwise the OTHER party
  1112→          // stays in the call forever waiting for a peer that's gone.
  1113→          //
  1114→          // We check 3 seconds after disconnect — this gives a brief window
  1115→          // for the client to reconnect (mobile network flaps) before we
  1116→          // declare the call dead. If the user reconnects within 3s, their
  1117→          // new socket re-registers and the call continues normally.
  1118→          setTimeout(() => {
  1119→            (async () => {
  1120→              try {
  1121→                // If the user reconnected (any socket is alive), skip cleanup.
  1122→                if (isUserOnline(userId)) return
  1123→
  1124→                // Find any active call where this user is caller OR recipient
  1125→                // AND the call is in an active state (accepted/connected).
  1126→                const activeCalls = await prisma.call.findMany({
  1127→                  where: {
  1128→                    OR: [{ callerId: userId }, { recipientId: userId }],
  1129→                    status: { in: ['accepted', 'ringing'] },
  1130→                  },
  1131→                  select: {
  1132→                    id: true,
  1133→                    callerId: true,
  1134→                    recipientId: true,
  1135→                    status: true,
  1136→                    startedAt: true,
  1137→                    conversationId: true,
  1138→                  },
  1139→                })
  1140→                for (const c of activeCalls) {
  1141→                  const duration = c.startedAt
  1142→                    ? Math.floor((Date.now() - c.startedAt.getTime()) / 1000)
  1143→                    : null
  1144→                  // ringing → missed (caller's perspective) or cancelled (recipient's)
  1145→                  // accepted → ended (with duration)
  1146→                  const newStatus = c.status === 'ringing'
  1147→                    ? (c.callerId === userId ? 'cancelled' : 'missed')
  1148→                    : 'ended'
  1149→                  await prisma.call.update({
  1150→                    where: { id: c.id },
  1151→                    data: { status: newStatus, endedAt: new Date(), duration },
  1152→                  })
  1153→                  const otherId = c.callerId === userId ? c.recipientId : c.callerId
  1154→                  emitToUserSockets(io, otherId, 'call:ended', {
  1155→                    callId: c.id,
  1156→                    reason: newStatus === 'missed' ? 'missed' : newStatus === 'cancelled' ? 'cancelled' : 'peer-disconnected',
  1157→                    duration,
  1158→                  })
  1159→                  // Also notify the disconnecting user (in case they reconnect
  1160→                  // — they should see the call ended)
  1161→                  emitToUserSockets(io, userId, 'call:ended', {
  1162→                    callId: c.id,
  1163→                    reason: newStatus === 'missed' ? 'missed' : newStatus === 'cancelled' ? 'cancelled' : 'peer-disconnected',
  1164→                    duration,
  1165→                  })
  1166→                }
  1167→              } catch (e) {
  1168→                console.error('[CALL] disconnect cleanup error:', e instanceof Error ? e.message : e)
  1169→              }
  1170→            })()
  1171→          }, 3_000).unref?.() // 3s grace period for reconnect
  1172→        }
  1173→      }
  1174→    })
  1175→  })
  1176→}
  1177→
  1178→async function broadcastPresenceToPartners(
  1179→  io: IoServer,
  1180→  userId: string,
  1181→  event: string,
  1182→  payload: any,
  1183→) {
  1184→  try {
  1185→    const participations = await prisma.conversationParticipant.findMany({
  1186→      where: { userId },
  1187→      select: { conversationId: true },
  1188→    })
  1189→    const roomSet = new Set<string>()
  1190→    for (const p of participations) {
  1191→      roomSet.add(`conversation:${p.conversationId}`)
  1192→    }
  1193→    for (const room of roomSet) {
  1194→      io.to(room).emit(event, payload)
  1195→    }
  1196→  } catch (e) {
  1197→    console.error('[SOCKET] broadcastPresenceToPartners error:', e)
  1198→  }
  1199→}
